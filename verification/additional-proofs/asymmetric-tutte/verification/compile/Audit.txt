def AsymmetricTutte.Region : ℝ → ℝ → Prop :=
fun x y =>
  1 < x ∧
    1 < y ∧
      (∀ (k : ℕ), 2 ≤ k → (√(x - 1) + ↑k * √(y - 1)) ^ 2 ≤ x * y ^ k) ∧
        ∀ (k : ℕ), 2 ≤ k → (√(y - 1) + ↑k * √(x - 1)) ^ 2 ≤ y * x ^ k
def AsymmetricMerinoWelsh.Universal : ℝ → ℝ → Prop :=
fun x y =>
  ∀ (α : Type) (M : Matroid α) [inst : M.Finite],
    Results.TutteThreshold.IsAdmissible M →
      Results.TutteThreshold.tutte M 1 1 ^ 2 ≤ Results.TutteThreshold.tutte M x 0 * Results.TutteThreshold.tutte M 0 y
def Results.TutteThreshold.IsAdmissible.{u_1} : {α : Type u_1} → Matroid α → Prop :=
fun {α} M => (∀ (e : α), ¬M.IsLoop e) ∧ ∀ (e : α), ¬M.IsColoop e
def Results.TutteThreshold.tutte.{u_1} : {α : Type u_1} → (M : Matroid α) → [M.Finite] → ℝ → ℝ → ℝ :=
fun {α} M [M.Finite] x y =>
  ∑ S ∈ ⋯.toFinset.powerset,
    (x - 1) ^ (M.eRank.toNat - Results.TutteThreshold.rk M ↑S) * (y - 1) ^ (S.card - Results.TutteThreshold.rk M ↑S)
AsymmetricTutte.universal_iff_region {x y : ℝ} (hx : 0 ≤ x) (hy : 0 ≤ y) :
  AsymmetricMerinoWelsh.Universal x y ↔ AsymmetricTutte.Region x y
