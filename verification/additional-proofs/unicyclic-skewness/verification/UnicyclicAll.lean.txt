theorem R_order {t z q : ℝ} (hz0 : 0 ≤ z) (hq0 : 0 ≤ q)
    (hz1 : z ≤ 1) (hq1 : q ≤ 1) (hzq : z < q)
    (hr : 0 ≤ R t z) : 0 < R t q := by
  have hd := D_pos hz0 hz1
  have hq := D_pos hq0 hq1
  have hz' : 0 ≤ 1-z := sub_nonneg.mpr hz1
  have hq' : 0 ≤ 1-q := sub_nonneg.mpr hq1
  have hE : 0 < 15 + (1-z)*(1-q)*(20+6*(z+q)) := by positivity
  have hid : D z * R t q - D q * R t z =
      (q-z)*(D z * D q + 105*(15+(1-z)*(1-q)*(20+6*(z+q)))) := by
    dsimp [R,D]
    ring
  have hprod : 0 <
      (q-z)*(D z * D q + 105*(15+(1-z)*(1-q)*(20+6*(z+q)))) := by
    exact mul_pos (sub_pos.mpr hzq) (add_pos (mul_pos hd hq) (mul_pos (by norm_num) hE))
  have hqr : 0 ≤ D q * R t z := mul_nonneg (le_of_lt hq) hr
  have : 0 < D z * R t q := by linarith
  exact pos_of_mul_pos_right this (le_of_lt hd)

theorem scalar_lower {t z : ℝ} (ht0 : 0 ≤ t) (ht1 : t ≤ 1)
    (hz0 : 0 ≤ z) (hz1 : z ≤ 1) (hr : 0 ≤ R t z) :
    0 ≤ F t z := by
  by_contra hf
  have hfn : F t z < 0 := lt_of_not_ge hf
  have hone : 0 ≤ F t 1 := by
    have hid : F t 1 = (2*t-1)^2*(12*t^2+4*t+1) := by
      dsimp [F]
      ring
    rw [hid]
    positivity
  have hb : 0 < 24*t^2-7 := by
    by_contra hbn
    have hbn' : 24*t^2-7 ≤ 0 := le_of_not_gt hbn
    have hm : 0 ≤ (z-1)*(24*t^2-7) :=
      mul_nonneg_of_nonpos_of_nonpos (sub_nonpos.mpr hz1) hbn'
    have hid : F t z = F t 1 + (z-1)*(24*t^2-7) := by
      dsimp [F]
      ring
    linarith
  let q : ℝ := -(48*t^4-32*t^3-24*t^2+8)/(24*t^2-7)
  have hroot : F t q = 0 := by
    dsimp [F,q]
    field_simp [ne_of_gt hb]
    ring
  have hdiff : F t z = (z-q)*(24*t^2-7) := by
    dsimp [F] at hroot ⊢
    nlinarith only [hroot]
  have hzq : z < q := by
    nlinarith only [hdiff, hfn, hb]
  have hq0 : 0 ≤ q := le_trans hz0 (le_of_lt hzq)
  have hdiff1 : F t 1 = (1-q)*(24*t^2-7) := by
    dsimp [F] at hroot ⊢
    nlinarith only [hroot]
  have hq1 : q ≤ 1 := by
    nlinarith only [hdiff1, hone, hb]
  have htl : (27/50 : ℝ) ≤ t := by
    by_contra hn
    have hn' : t < (27/50 : ℝ) := lt_of_not_ge hn
    have hh : 0 < ((27/50 : ℝ)-t)*((27/50 : ℝ)+t) :=
      mul_pos (sub_pos.mpr hn') (by linarith)
    nlinarith only [hh,hb]
  have hp : 0 ≤ P t := P_nonneg htl ht1
  have hid : (24*t^2-7)^4 * R t q = 2*(t-1)*P t := by
    dsimp [q,R,D,P]
    field_simp [ne_of_gt hb]
    ring

  have hleft : 0 < (24*t^2-7)^4 * R t q := mul_pos (pow_pos hb 4) hrq
  have hright : 2*(t-1)*P t ≤ 0 :=
    mul_nonpos_of_nonpos_of_nonneg (by linarith) hp

def gap (r : ℝ) : ℝ :=
  2*r + (2/3 : ℝ)*r^3 + (2/5 : ℝ)*r^5 +
    2*r^7/(7*(1-r^2)) - Real.log (1+r) + Real.log (1-r)

theorem gap_deriv {r : ℝ} (hr0 : 0 ≤ r) (hr1 : r < 1) :
    HasDerivAt gap (4*r^8/(7*(1-r^2)^2)) r := by
  have hp : 0 < 1+r := by linarith
  have hm : 0 < 1-r := by linarith
  have hsq : 0 < 1-r^2 := by nlinarith
  have hd : 7*(1-r^2) ≠ 0 := ne_of_gt (mul_pos (by norm_num) hsq)
  have hi := hasDerivAt_id r
  have hd' : HasDerivAt (fun x : ℝ => 7*(1-x^2)) (-14*r) r := by
    convert (((hasDerivAt_const r (1 : ℝ)).sub (hi.pow 2)).const_mul 7) using 1 ;
      simp only [id_eq] ; ring
  have hq := ((hi.pow 7).const_mul 2).div hd' hd
  have hlp := (hi.const_add 1).log (ne_of_gt hp)
  have hlm := ((hasDerivAt_const r (1 : ℝ)).sub hi).log (ne_of_gt hm)
  have ht := (((((hi.const_mul 2).add ((hi.pow 3).const_mul (2/3 : ℝ))).add
    ((hi.pow 5).const_mul (2/5 : ℝ))).add hq).sub hlp).add hlm
  convert ht using 1 ; simp only [id_eq] ;
    field_simp [ne_of_gt hp, ne_of_gt hm, ne_of_gt hsq] ; ring

theorem gap_nonneg {r : ℝ} (hr0 : 0 ≤ r) (hr1 : r < 1) : 0 ≤ gap r := by
  have hh : ∀ x ∈ Set.Icc (0 : ℝ) r,
      HasDerivAt gap (4*x^8/(7*(1-x^2)^2)) x := by
    intro x hx
    exact gap_deriv hx.1 (lt_of_le_of_lt hx.2 hr1)
  have hc : ContinuousOn gap (Set.Icc (0 : ℝ) r) := by
    intro x hx
    exact (hh x hx).continuousAt.continuousWithinAt
  have hm : MonotoneOn gap (Set.Icc (0 : ℝ) r) :=
    monotoneOn_of_hasDerivWithinAt_nonneg (convex_Icc 0 r) hc
      (fun x hx => (hh x (interior_subset hx)).hasDerivWithinAt)
      (fun x _ => by positivity)
  have h := hm ⟨le_rfl, hr0⟩ ⟨hr0, le_rfl⟩ hr0
  simpa [gap] using h

theorem log_ratio_upper {r : ℝ} (hr0 : 0 ≤ r) (hr1 : r < 1) :
    Real.log ((1+r)/(1-r)) ≤
      2*r+(2/3 : ℝ)*r^3+(2/5 : ℝ)*r^5+2*r^7/(7*(1-r^2)) := by
  have hp : 1+r ≠ 0 := by linarith
  have hm : 1-r ≠ 0 := by linarith
  rw [Real.log_div hp hm]
  have h := gap_nonneg hr0 hr1
  dsimp [gap] at h
  linarith

theorem rational_constraint {r t : ℝ} (hr0 : 0 < r) (hr1 : r < 1)
    (htur : (1-r^2)/2 + r/Real.log ((1+r)/(1-r)) ≤ t^2) :
    0 ≤ UnicyclicScalar.R t (r^2) := by
  have hm : 0 < 1-r := by linarith
  have hs : 0 < 1-r^2 := by nlinarith
  have hz0 : 0 ≤ r^2 := sq_nonneg r
  have hz1 : r^2 ≤ 1 := by nlinarith
  have hd : 0 < UnicyclicScalar.D (r^2) := UnicyclicScalar.D_pos hz0 hz1
  let L := Real.log ((1+r)/(1-r))
  have hL : 0 < L := by
    apply Real.log_pos
    apply (one_lt_div hm).mpr
    linarith
  have hu := UnicyclicLog.log_ratio_upper (le_of_lt hr0) hr1
  have heq :
      2*r+(2/3 : ℝ)*r^3+(2/5 : ℝ)*r^5+2*r^7/(7*(1-r^2)) =
        2*r*UnicyclicScalar.D (r^2)/(105*(1-r^2)) := by
    dsimp [UnicyclicScalar.D]
    field_simp [ne_of_gt hs]
    ring
  rw [heq] at hu
  have hU : L * (105*(1-r^2)) ≤ 2*r*UnicyclicScalar.D (r^2) :=
    (le_div_iff₀ (mul_pos (by norm_num) hs)).mp hu
  have hT : 2*r ≤ L*(2*t^2-1+r^2) := by
    have hdiv : r/L ≤ t^2-(1-r^2)/2 := by
      change (1-r^2)/2 + r/L ≤ t^2 at htur
      linarith only [htur]
    have hh := (div_le_iff₀ hL).mp hdiv
    nlinarith only [hh]

  have hid : L * UnicyclicScalar.R t (r^2) =
      L*(2*t^2-1+r^2)*UnicyclicScalar.D (r^2) - L*(105*(1-r^2)) := by
    dsimp [UnicyclicScalar.R]
    ring
  have hh : 0 ≤ L * UnicyclicScalar.R t (r^2) := by
    linarith only [hTD,hU,hid]

theorem third_elementary_nonneg (s : Finset ι) (x : ι → ℝ)
    (hx : ∀ i ∈ s, 0 ≤ x i) :
    0 ≤ (∑ i ∈ s, x i)^3 -
      3 * (∑ i ∈ s, x i) * (∑ i ∈ s, (x i)^2) +
      2 * (∑ i ∈ s, (x i)^3) := by
  classical
  induction s using Finset.induction_on with
  | empty => simp
  | @insert a s ha ih =>
    have hxs : ∀ i ∈ s, 0 ≤ x i := fun i hi => hx i (Finset.mem_insert_of_mem hi)
    have hxa : 0 ≤ x a := hx a (Finset.mem_insert_self a s)
    have hprev := ih hxs
    have hsq := Finset.sum_sq_le_sq_sum_of_nonneg hxs
    have hprod : 0 ≤ 3 * x a *
        ((∑ i ∈ s, x i)^2 - ∑ i ∈ s, (x i)^2) := by
      exact mul_nonneg (by positivity) (sub_nonneg.mpr hsq)
    simp only [Finset.sum_insert ha]
    nlinarith only [hprev, hprod]

theorem simplex_second_moment_le_one (s : Finset ι) (x : ι → ℝ)
    (hx : ∀ i ∈ s, 0 ≤ x i) (hsum : ∑ i ∈ s, x i = 1) :
    (∑ i ∈ s, (x i)^2) ≤ 1 := by
  have h := Finset.sum_sq_le_sq_sum_of_nonneg hx
  simpa only [hsum, one_pow] using h

theorem simplex_third_relation (s : Finset ι) (x : ι → ℝ)
    (hx : ∀ i ∈ s, 0 ≤ x i) (hsum : ∑ i ∈ s, x i = 1) :
    0 ≤ 1 - 3 * (∑ i ∈ s, (x i)^2) + 2 * (∑ i ∈ s, (x i)^3) := by
  simpa only [hsum, one_pow, mul_one] using third_elementary_nonneg s x hx

theorem sum_cubes_le_sqrt_second_cube (s : Finset ι) (x : ι → ℝ)
    (t : ℝ) (ht : 0 ≤ t) (hx : ∀ i ∈ s, 0 ≤ x i)
    (htsq : ∑ i ∈ s, (x i)^2 = t^2) :
    (∑ i ∈ s, (x i)^3) ≤ t^3 := by
  have hpoint : ∀ i ∈ s, (x i)^3 ≤ t * (x i)^2 := by
    intro i hi
    have hxi := hx i hi
    have hsqi : (x i)^2 ≤ t^2 := by
      rw [← htsq]
      exact Finset.single_le_sum (fun j _ => sq_nonneg (x j)) hi
    have hle : x i ≤ t := by nlinarith
    have hmul := mul_le_mul_of_nonneg_right hle (sq_nonneg (x i))
    nlinarith only [hmul]
  calc
    (∑ i ∈ s, (x i)^3) ≤ ∑ i ∈ s, t * (x i)^2 :=
      Finset.sum_le_sum hpoint
    _ = t * ∑ i ∈ s, (x i)^2 := by rw [Finset.mul_sum]
    _ = t^3 := by rw [htsq]; ring

    (hc2 : 0 ≤ c2) (hp2 : p2 ≤ 1)
    (he3 : 0 ≤ 1 - 3*p2 + 2*p3)
    (hvar : 0 ≤ (1-c2)/2 + p2*c2) :
    c2 * (3*p2^2 - 2*p3) + (c2-1)*(1-3*p2)/2 ≤ 1 := by
  let e2 := (1-p2)/2

  have hbound : 4*c2*e2 ≤ c2+1 := by dsimp [e2]; nlinarith only [hvar]
  have hprod := mul_le_mul_of_nonneg_right hbound he2
  have he3prod := mul_nonneg hc2 he3
  dsimp [e2] at hprod
  nlinarith only [hprod, he3prod]

    (hx : ∀ i ∈ s, 0 ≤ x i) (hsum : ∑ i ∈ s, x i = 1)
    (hvar : 0 ≤ (1-c^2)/2 + (∑ i ∈ s, (x i)^2)*c^2) :
    c^2 * (3*(∑ i ∈ s, (x i)^2)^2 - 2*(∑ i ∈ s, (x i)^3)) +
      (c^2-1)*(1-3*(∑ i ∈ s, (x i)^2))/2 ≤ 1 := by
  exact skewness_upper_scalar _ _ _ (sq_nonneg c)
    (simplex_second_moment_le_one s x hx hsum)
    (simplex_third_relation s x hx hsum) hvar

theorem linear_mul_coeff_zero (a : ℝ) (P : Polynomial ℝ) :
    ((1+C a*X)*P).coeff 0 = P.coeff 0 := by
  simp [add_mul, mul_assoc, coeff_C_mul]

    ((1+C a*X)*P).coeff (n+1) = P.coeff (n+1) + a*P.coeff n := by
  simp [add_mul, mul_assoc, coeff_C_mul, coeff_X_mul]

theorem linearProduct_coefficients (s : Finset ι) (x : ι → ℝ) :
    (linearProduct s x).coeff 0 = 1 ∧
    (linearProduct s x).coeff 1 = ∑ i ∈ s, x i ∧
    2*(linearProduct s x).coeff 2 =
      (∑ i ∈ s, x i)^2 - ∑ i ∈ s, (x i)^2 ∧

  induction s using Finset.induction_on with
  | empty => simp [linearProduct, coeff_one]
  | @insert a s ha ih =>
    obtain ⟨h0,h1,h2,h3⟩ := ih
    have hp : linearProduct (insert a s) x =
        (1+C (x a)*X)*linearProduct s x := by
      simp [linearProduct, Finset.prod_insert ha]
    rw [hp]
    simp only [linear_mul_coeff_zero, linear_mul_coeff_succ,
      Finset.sum_insert ha]
    norm_num only [Nat.reduceAdd]
    refine ⟨h0, ?_, ?_, ?_⟩
    · rw [h0, h1]
      ring
    · rw [h1]
      nlinarith only [h2]
    · have hm := congrArg (fun r : ℝ => 3*x a*r) h2
      dsimp at hm
      nlinarith only [h3, hm]

/-!
Conversion of a positive-spectrum factorization into characteristic
coefficients. This is a polynomial theorem. The matrix factorization itself
is not assumed as a definition and must be supplied by the matrix proof.
-/

namespace UnicyclicProduct

open scoped BigOperators
open Polynomial

variable {ι : Type*}

noncomputable def shiftedProduct (s : Finset ι) (theta : ι → ℝ) : Polynomial ℝ :=
  ∏ i ∈ s, (X+C (theta i))

theorem shiftedProduct_factorization (s : Finset ι) (theta : ι → ℝ)
    (htheta : ∀ i ∈ s, theta i ≠ 0) :
    shiftedProduct s theta =
      C (∏ i ∈ s, theta i) * linearProduct s (fun i => 1/theta i) := by
  have hpoint : ∀ i ∈ s,
      X+C (theta i) = C (theta i)*(1+C (1/theta i)*X) := by
    intro i hi
    have hprod : C (theta i)*C (1/theta i) = (1 : Polynomial ℝ) := by
      rw [← map_mul]
      simp [one_div, htheta i hi]
    rw [mul_add, mul_one, ← mul_assoc, hprod, one_mul]
    ring
  dsimp [shiftedProduct, linearProduct]
  calc
    (∏ i ∈ s, (X+C (theta i))) =
        ∏ i ∈ s, C (theta i)*(1+C (1/theta i)*X) := Finset.prod_congr rfl hpoint
    _ = (∏ i ∈ s, C (theta i)) * (∏ i ∈ s, (1+C (1/theta i)*X)) :=
      Finset.prod_mul_distrib
    _ = C (∏ i ∈ s, theta i) * (∏ i ∈ s, (1+C (1/theta i)*X)) := by
      rw [map_prod]

theorem characteristic_coefficients_from_spectrum
    (s : Finset ι) (theta : ι → ℝ) (P : Polynomial ℝ) (d : ℝ)
    (htheta : ∀ i ∈ s, theta i ≠ 0)

    P.coeff 1 = d*(∑ i ∈ s, 1/theta i) ∧
    P.coeff 2 = d*((∑ i ∈ s, 1/theta i)^2 - ∑ i ∈ s, (1/theta i)^2)/2 ∧
    P.coeff 3 = d*((∑ i ∈ s, 1/theta i)^3 -

      2*(∑ i ∈ s, (1/theta i)^3))/6 := by

  have hc2 := congrArg (fun Q : Polynomial ℝ => Q.coeff 2) hid
  have hc3 := congrArg (fun Q : Polynomial ℝ => Q.coeff 3) hid

  refine ⟨?_, ?_, ?_⟩
  · rw [h1] at hc1
    exact hc1
  · have hm := congrArg (fun r : ℝ => d*r) h2
    dsimp at hm
    rw [hc2]
    nlinarith only [hm]
  · have hm := congrArg (fun r : ℝ => d*r) h3
    dsimp at hm
    rw [hc3]
    nlinarith only [hm]

/- proof/CumulantAlgebra.lean; SHA256 81f20d38bc87e43e1d0026c49832209284053d562a4e423e3a89eb77d5889c58 -/
section CheckedModule7

/-!
Algebra of the first three winding-current cumulants.

If P(s) = a1*s + a2*s^2 + a3*s^3 + ... is the characteristic
polynomial at zero tilt, implicit differentiation of
P(lambda(k)) = U*(exp(k)-1) + V*(exp(-k)-1)
gives the three displayed equations proved here for the explicit definitions.

This file verifies their algebraic consequences. It does NOT prove that a
given Markov generator has this tilted characteristic polynomial, nor does
it prove the spectral coefficient assumptions from the transition rates.
-/

namespace UnicyclicCumulant

noncomputable section

def first (a1 U V : ℝ) : ℝ := (U-V)/a1

def second (a1 a2 U V : ℝ) : ℝ :=
  ((U+V) - 2*a2*(first a1 U V)^2)/a1

def third (a1 a2 a3 U V : ℝ) : ℝ :=
  ((U-V) - 6*a2*(first a1 U V)*(second a1 a2 U V) -
    6*a3*(first a1 U V)^3)/a1

theorem first_equation (a1 U V : ℝ) (ha1 : a1 ≠ 0) :
    a1 * first a1 U V = U-V := by
  dsimp [first]
  field_simp

theorem second_equation (a1 a2 U V : ℝ) (ha1 : a1 ≠ 0) :
    a1 * second a1 a2 U V + 2*a2*(first a1 U V)^2 = U+V := by
  dsimp [second]
  rw [mul_div_cancel₀ _ ha1]
  ring

theorem third_equation (a1 a2 a3 U V : ℝ) (ha1 : a1 ≠ 0) :
    a1 * third a1 a2 a3 U V +
      6*a2*(first a1 U V)*(second a1 a2 U V) +
      6*a3*(first a1 U V)^3 = U-V := by
  dsimp [third]
  rw [mul_div_cancel₀ _ ha1]
  ring

theorem scaled_cumulant_reduction
    (a1 a2 a3 U V d T c p2 p3 C1 C2 C3 : ℝ)
    (hd : d ≠ 0)
    (ha1 : a1 = d*T)
    (ha2 : a2 = d*T^2*(1-p2)/2)
    (ha3 : a3 = d*T^3*(1-3*p2+2*p3)/6)
    (hminus : U-V = d*c)
    (hplus : U+V = d*(c^2+1)/2)
    (h1 : a1*C1 = U-V)
    (h2 : a1*C2 + 2*a2*C1^2 = U+V)
    (h3 : a1*C3 + 6*a2*C1*C2 + 6*a3*C1^3 = U-V) :
    T*C1 = c ∧
    T*C2 = (1-c^2)/2 + p2*c^2 ∧
    T*C3 = c*(c^2*(3*p2^2-2*p3) + (c^2-1)*(1-3*p2)/2) := by
  have hC1 : T*C1 = c := by
    apply mul_left_cancel₀ hd
    calc
      d*(T*C1) = a1*C1 := by rw [ha1]; ring
      _ = U-V := h1
      _ = d*c := hminus
  have hC2sum : T*C2 + (1-p2)*c^2 = (c^2+1)/2 := by
    apply mul_left_cancel₀ hd
    calc
      d*(T*C2 + (1-p2)*c^2) = a1*C2 + 2*a2*C1^2 := by
        rw [ha1, ha2, ← hC1]
        ring
      _ = U+V := h2
      _ = d*((c^2+1)/2) := by rw [hplus]; ring
  have hC2 : T*C2 = (1-c^2)/2 + p2*c^2 := by linarith only [hC2sum]
  have hC3sum : T*C3 + 3*(1-p2)*c*((1-c^2)/2+p2*c^2) +
      (1-3*p2+2*p3)*c^3 = c := by
    apply mul_left_cancel₀ hd
    calc
      d*(T*C3 + 3*(1-p2)*c*((1-c^2)/2+p2*c^2) +
          (1-3*p2+2*p3)*c^3) =
          a1*C3 + 6*a2*C1*C2 + 6*a3*C1^3 := by
        rw [ha1, ha2, ha3, ← hC2, ← hC1]
        ring
      _ = U-V := h3
      _ = d*c := hminus
  refine ⟨hC1, hC2, ?_⟩
  nlinarith only [hC3sum]

theorem cumulant_ratio_reduction
    (a1 a2 a3 U V d T c p2 p3 C1 C2 C3 : ℝ)
    (hd : d ≠ 0) (hT : T ≠ 0) (hc : c ≠ 0)
    (ha1 : a1 = d*T)
    (ha2 : a2 = d*T^2*(1-p2)/2)
    (ha3 : a3 = d*T^3*(1-3*p2+2*p3)/6)
    (hminus : U-V = d*c)
    (hplus : U+V = d*(c^2+1)/2)
    (h1 : a1*C1 = U-V)
    (h2 : a1*C2 + 2*a2*C1^2 = U+V)
    (h3 : a1*C3 + 6*a2*C1*C2 + 6*a3*C1^3 = U-V) :
    C2/C1 = c*p2 - (c^2-1)/(2*c) ∧
    C3/C1 = c^2*(3*p2^2-2*p3) + (c^2-1)*(1-3*p2)/2 := by
  obtain ⟨hC1, hC2, hC3⟩ := scaled_cumulant_reduction
    a1 a2 a3 U V d T c p2 p3 C1 C2 C3 hd ha1 ha2 ha3
    hminus hplus h1 h2 h3
  have hC1nz : C1 ≠ 0 := by
    intro hz
    rw [hz, mul_zero] at hC1
    exact hc hC1.symm
  constructor
  · have hratio : C2/C1 = ((1-c^2)/2+p2*c^2)/c := by
      apply (div_eq_div_iff hC1nz hc).2
      calc
        C2*c = C2*(T*C1) := by rw [hC1]
        _ = (T*C2)*C1 := by ring
        _ = ((1-c^2)/2+p2*c^2)*C1 := by rw [hC2]
    rw [hratio]
    field_simp
    ring
  · apply (div_eq_iff hC1nz).2
    apply mul_left_cancel₀ hT
    calc
      T*C3 = c*(c^2*(3*p2^2-2*p3) + (c^2-1)*(1-3*p2)/2) := hC3
      _ = T*((c^2*(3*p2^2-2*p3) + (c^2-1)*(1-3*p2)/2)*C1) := by
        rw [← hC1]
        ring

theorem explicit_cumulant_ratio_reduction
    (a1 a2 a3 U V d T c p2 p3 : ℝ)
    (hd : d ≠ 0) (hT : T ≠ 0) (hc : c ≠ 0)
    (ha1 : a1 = d*T)
    (ha2 : a2 = d*T^2*(1-p2)/2)
    (ha3 : a3 = d*T^3*(1-3*p2+2*p3)/6)
    (hminus : U-V = d*c)
    (hplus : U+V = d*(c^2+1)/2) :
    second a1 a2 U V / first a1 U V = c*p2 - (c^2-1)/(2*c) ∧
    third a1 a2 a3 U V / first a1 U V =
      c^2*(3*p2^2-2*p3) + (c^2-1)*(1-3*p2)/2 := by
  have ha1nz : a1 ≠ 0 := by rw [ha1]; exact mul_ne_zero hd hT
  exact cumulant_ratio_reduction a1 a2 a3 U V d T c p2 p3
    (first a1 U V) (second a1 a2 U V) (third a1 a2 a3 U V)
    hd hT hc ha1 ha2 ha3 hminus hplus
    (first_equation a1 U V ha1nz)
    (second_equation a1 a2 U V ha1nz)
    (third_equation a1 a2 a3 U V ha1nz)

#print axioms scaled_cumulant_reduction
#print axioms explicit_cumulant_ratio_reduction

end
end UnicyclicCumulant

end CheckedModule7

/-!
The characteristic-polynomial bound, with a positive spectral shift and the
thermodynamic uncertainty inequality explicit as hypotheses. This is not yet
the arbitrary-rate Markov theorem: the separate matrix and corrector modules
must supply these hypotheses from rates before that conclusion can be claimed.
-/

theorem bound_from_spectral_shift_and_uncertainty {ι : Type*}
    (s : Finset ι) (hs : s.Nonempty) (theta : ι → ℝ)
    (P : Polynomial ℝ) (a b : ℝ) (ha : 0 < a) (hb : 0 < b) (hba : b < a)
    (hth : ∀ i ∈ s, 0 < theta i)
    (hd : (a-b)^2 = ∏ i ∈ s, theta i)
    (hfactor : P+C ((a-b)^2) = UnicyclicProduct.shiftedProduct s theta)
    (htur : 1/Real.log (a/b) ≤
      second (P.coeff 1) (P.coeff 2) (a^2) (b^2) /
        first (P.coeff 1) (a^2) (b^2)) :
    -(1/16 : ℝ) ≤
      third (P.coeff 1) (P.coeff 2) (P.coeff 3) (a^2) (b^2) /
        first (P.coeff 1) (a^2) (b^2) ∧
      third (P.coeff 1) (P.coeff 2) (P.coeff 3) (a^2) (b^2) /
        first (P.coeff 1) (a^2) (b^2) ≤ 1 := by
  let T := ∑ i ∈ s, 1/theta i
  have hT : 0 < T := Finset.sum_pos (fun i hi => one_div_pos.mpr (hth i hi)) hs
  let x := fun i => (1/theta i)/T
  let p2 := ∑ i ∈ s, (x i)^2
  let p3 := ∑ i ∈ s, (x i)^3
  let d := (a-b)^2
  let c := (a+b)/(a-b)
  let r := (a-b)/(a+b)
  have hab : 0 < a+b := by linarith
  have hdiff : 0 < a-b := sub_pos.mpr hba
  have hd0 : d ≠ 0 := ne_of_gt (sq_pos_of_pos hdiff)
  have hc0 : c ≠ 0 := div_ne_zero (ne_of_gt hab) (ne_of_gt hdiff)
  have hr0 : 0 < r := div_pos hdiff hab
  have hr1 : r < 1 := by
    apply (div_lt_one hab).mpr
    linarith
  have hcr : c = 1/r := by
    dsimp [c,r]
    field_simp
  have hx : ∀ i ∈ s, 0 ≤ x i := by
    intro i hi
    exact le_of_lt (div_pos (one_div_pos.mpr (hth i hi)) hT)
  have hsum : ∑ i ∈ s, x i = 1 := by
    dsimp [x]
    rw [← Finset.sum_div]
    exact div_self (ne_of_gt hT)
  obtain ⟨ha1,ha2,ha3⟩ := UnicyclicProduct.normalized_characteristic_coefficients_from_spectrum
    s theta P d T (fun i hi => ne_of_gt (hth i hi)) hd hfactor rfl (ne_of_gt hT)
  have hminus : a^2-b^2 = d*c := by
    dsimp [d,c]
    field_simp
    ring
  have hplus : a^2+b^2 = d*(c^2+1)/2 := by
    dsimp [d,c]
    field_simp
    ring
  obtain ⟨hF,hS⟩ := explicit_cumulant_ratio_reduction
    (P.coeff 1) (P.coeff 2) (P.coeff 3) (a^2) (b^2) d T c p2 p3
    hd0 (ne_of_gt hT) hc0 ha1 ha2 ha3 hminus hplus
  have hF' : c*p2-(c^2-1)/(2*c) = (p2-(1-r^2)/2)/r := by
    rw [hcr]
    field_simp [ne_of_gt hr0]
    ring
  rw [hF,hF'] at htur
  have heq : (1+r)/(1-r) = a/b := by
    apply (div_eq_div_iff (ne_of_gt (sub_pos.mpr hr1)) (ne_of_gt hb)).mpr
    dsimp [r]
    field_simp
    ring
  have htur' : (1-r^2)/2+r/Real.log ((1+r)/(1-r)) ≤ p2 := by
    rw [heq]
    have h := (le_div_iff₀ hr0).mp htur
    have he : 1/Real.log (a/b)*r = r/Real.log (a/b) := by ring
    rw [he] at h
    nlinarith only [h]
  have hbounds := UnicyclicMoments.spectral_skewness_interval s x r hr0 hr1 hx hsum htur'
  have hS' : c^2*(3*p2^2-2*p3)+(c^2-1)*(1-3*p2)/2 =
      UnicyclicMoments.skewness p2 p3 r := by
    rw [hcr]
    dsimp [UnicyclicMoments.skewness]
    field_simp [ne_of_gt hr0]
    ring
  rw [hS,hS']
  exact hbounds

#print axioms bound_from_spectral_shift_and_uncertainty
end

/-!
The arbitrary-size combinatorial and determinant bridge for a tilted cycle.
This is an auxiliary development; it does not yet state or prove the original
Markov-current inequality. No hypothesis is intended to assume the determinant
identity itself.

open scoped BigOperators
open Equiv

namespace UnicyclicDeterminant

noncomputable section

variable {ι : Type*} [Fintype ι] [DecidableEq ι]

omit [Fintype ι] [DecidableEq ι] in
theorem cycle_no_fixed [Fintype ι] [DecidableEq ι] (σ : Perm ι)
    (hc : σ.IsCycleOn (Finset.univ : Finset ι))
    (hn : 3 ≤ Fintype.card ι) (i : ι) : σ i ≠ i := by
  intro h
  have hd : Fintype.card ι ∣ 1 := by
    simpa using (hc.pow_apply_eq (Finset.mem_univ i) (n := 1)).mp (by simpa using h)
  have := Nat.le_of_dvd (by decide : 0 < 1) hd
  omega

omit [Fintype ι] [DecidableEq ι] in
theorem cycle_no_two [Fintype ι] [DecidableEq ι] (σ : Perm ι)
    (hc : σ.IsCycleOn (Finset.univ : Finset ι))
    (hn : 3 ≤ Fintype.card ι) (i : ι) : σ (σ i) ≠ i := by
  intro h
  have hd : Fintype.card ι ∣ 2 := by
    apply (hc.pow_apply_eq (Finset.mem_univ i) (n := 2)).mp
    simpa [pow_two, Perm.mul_apply] using h
  have := Nat.le_of_dvd (by decide : 0 < 2) hd
  omega

theorem unpaired_forward_forces_cycle (σ π : Perm ι)
    (hc : σ.IsCycleOn (Finset.univ : Finset ι))
    (hn : 3 ≤ Fintype.card ι)
    (hs : ∀ i, π i = i ∨ π i = σ i ∨ π i = σ⁻¹ i)
    (i : ι) (hi : π i = σ i) (hnext : π (σ i) ≠ i) : π = σ := by
  have step (j : ι) (hj : π j = σ j ∧ π (σ j) ≠ j) :
      π (σ j) = σ (σ j) ∧ π (σ (σ j)) ≠ σ j := by
    have hforward : π (σ j) = σ (σ j) := by
      rcases hs (σ j) with hfix | hforward | hback
      · have hh : σ j = j := π.injective (hfix.trans hj.1.symm)
        exact False.elim (cycle_no_fixed σ hc hn j hh)
      · exact hforward
      · exact False.elim (hj.2 (by simpa using hback))
    refine ⟨hforward, ?_⟩
    intro hbad
    have hh : σ (σ j) = j := π.injective (hbad.trans hj.1.symm)
    exact cycle_no_two σ hc hn j hh
  have propagate : ∀ k : ℕ,
      π ((σ ^ k) i) = σ ((σ ^ k) i) ∧
        π (σ ((σ ^ k) i)) ≠ (σ ^ k) i := by
    intro k
    induction k with
    | zero => simpa using And.intro hi hnext
    | succ k ih => simpa [pow_succ', Perm.mul_apply] using step ((σ ^ k) i) ih
  apply Equiv.ext
  intro j
  obtain ⟨k, _, hk⟩ := hc.exists_pow_eq (Finset.mem_univ i) (Finset.mem_univ j)
  simpa [hk] using (propagate k).1

theorem counted_edge_paired (σ π : Perm ι)
    (hc : σ.IsCycleOn (Finset.univ : Finset ι))
    (hn : 3 ≤ Fintype.card ι)
    (hs : ∀ i, π i = i ∨ π i = σ i ∨ π i = σ⁻¹ i)
    (hforward : π ≠ σ) (hback : π ≠ σ⁻¹) (i : ι) :
    π i = σ i ↔ π (σ i) = i := by
  constructor
  · intro hi
    by_contra hnext
    exact hforward (unpaired_forward_forces_cycle σ π hc hn hs i hi hnext)
  · intro hi
    by_contra hprev
    have hs' : ∀ j, π j = j ∨ π j = σ⁻¹ j ∨ π j = (σ⁻¹)⁻¹ j := by
      intro j
      simpa [or_comm, or_left_comm, or_assoc] using hs j
    apply hback
    apply unpaired_forward_forces_cycle σ⁻¹ π hc.inv hn hs' (σ i)
    · simpa using hi
    · simpa using hprev

def edgeTilt (A : Matrix ι ι ℝ) (a b : ι) (q : ℝ) : Matrix ι ι ℝ :=
  fun i j =>
    (if j = a then if i = b then q else 1 else 1) *
    (if j = b then if i = a then q⁻¹ else 1 else 1) * A i j

theorem edgeTilt_prod (A : Matrix ι ι ℝ) (a b : ι) (q : ℝ) (π : Perm ι) :
    (∏ j, edgeTilt A a b q (π j) j) =
      (if π a = b then q else 1) * (if π b = a then q⁻¹ else 1) *
        ∏ j, A (π j) j := by
  unfold edgeTilt
  rw [Finset.prod_mul_distrib, Finset.prod_mul_distrib]
  simp only [Finset.prod_ite_eq', Finset.mem_univ, ite_true]

theorem permutation_term_tilt (A : Matrix ι ι ℝ) (σ π : Perm ι)
    (hc : σ.IsCycleOn (Finset.univ : Finset ι))
    (hn : 3 ≤ Fintype.card ι)
    (hA : ∀ i j, i ≠ j → i ≠ σ j → i ≠ σ⁻¹ j → A i j = 0)
    (a : ι) (q : ℝ) (hq : q ≠ 0) :
    (∏ j, edgeTilt A a (σ a) q (π j) j) =
      (∏ j, A (π j) j) +
      (if π = σ then (q - 1) * ∏ j, A (π j) j else 0) +
      (if π = σ⁻¹ then (q⁻¹ - 1) * ∏ j, A (π j) j else 0) := by
  rw [edgeTilt_prod]
  have hinv : σ ≠ σ⁻¹ := by
    intro h
    have hh := congrArg (fun f : Perm ι => f (σ a)) h
    exact cycle_no_two σ hc hn a (by simpa using hh)
  by_cases hp : π = σ
  · subst π
    simp only [ite_true, hinv, ite_false, add_zero,
      cycle_no_two σ hc hn a, mul_one]
    ring
  by_cases hp' : π = σ⁻¹
  · subst π
    have hrev : σ⁻¹ a ≠ σ a := by
      intro h
      have hh := congrArg σ h
      exact cycle_no_two σ hc hn a (by simpa using hh.symm)
    simp only [hp, ite_false, add_zero, ite_true, hrev, one_mul,
      Perm.inv_apply_self]
    ring
  simp only [hp, hp', ite_false, add_zero]
  by_cases hz : (∏ j, A (π j) j) = 0
  · simp [hz]
  have hs : ∀ j, π j = j ∨ π j = σ j ∨ π j = σ⁻¹ j := by
    intro j
    by_contra h
    push_neg at h
    have he : A (π j) j = 0 := hA (π j) j h.1 h.2.1 h.2.2
    exact hz (Finset.prod_eq_zero (Finset.mem_univ j) he)
  have hpair := counted_edge_paired σ π hc hn hs hp hp' a
  by_cases he : π a = σ a
  · simp [he, hpair.mp he, hq]
  · have he' : π (σ a) ≠ a := fun h => he (hpair.mpr h)
    simp [he, he']

theorem cyclic_determinant_tilt (A : Matrix ι ι ℝ) (σ : Perm ι)
    (hc : σ.IsCycleOn (Finset.univ : Finset ι))
    (hn : 3 ≤ Fintype.card ι)
    (hA : ∀ i j, i ≠ j → i ≠ σ j → i ≠ σ⁻¹ j → A i j = 0)
    (a : ι) (q : ℝ) (hq : q ≠ 0) :
    (edgeTilt A a (σ a) q).det = A.det +
      (q - 1) * ((Perm.sign σ : ℤ) : ℝ) * (∏ j, A (σ j) j) +
      (q⁻¹ - 1) * ((Perm.sign σ⁻¹ : ℤ) : ℝ) * (∏ j, A (σ⁻¹ j) j) := by
  have hterm (π : Perm ι) :
      ((Perm.sign π : ℤ) : ℝ) * (∏ j, edgeTilt A a (σ a) q (π j) j) =
        ((Perm.sign π : ℤ) : ℝ) * (∏ j, A (π j) j) +
        (if π = σ then (q - 1) * ((Perm.sign σ : ℤ) : ℝ) *
          (∏ j, A (σ j) j) else 0) +
        (if π = σ⁻¹ then (q⁻¹ - 1) * ((Perm.sign σ⁻¹ : ℤ) : ℝ) *
          (∏ j, A (σ⁻¹ j) j) else 0) := by
    rw [permutation_term_tilt A σ π hc hn hA a q hq]
    split_ifs <;> simp_all <;> ring
  simp_rw [Matrix.det_apply', hterm]
  rw [Finset.sum_add_distrib, Finset.sum_add_distrib]
  simp

theorem one_way_permutation (σ π : Perm ι)
    (hc : σ.IsCycleOn (Finset.univ : Finset ι))
    (hn : 3 ≤ Fintype.card ι)
    (hs : ∀ i, π i = i ∨ π i = σ i) (hp : π ≠ 1) : π = σ := by
  have hex : ∃ i, π i ≠ i := by
    by_contra h
    push_neg at h
    exact hp (Equiv.ext h)
  obtain ⟨i, hi⟩ := hex
  have hforward : π i = σ i := (hs i).resolve_left hi
  have hnext : π (σ i) ≠ i := by
    intro h
    rcases hs (σ i) with hf | hf
    · exact cycle_no_fixed σ hc hn i (hf.symm.trans h)
    · exact cycle_no_two σ hc hn i (hf.symm.trans h)
  exact unpaired_forward_forces_cycle σ π hc hn
    (fun j => (hs j).imp_right Or.inl) i hforward hnext

theorem cyclic_bidiagonal_determinant (A : Matrix ι ι ℝ) (σ : Perm ι)
    (hc : σ.IsCycleOn (Finset.univ : Finset ι))
    (hn : 3 ≤ Fintype.card ι)
    (hA : ∀ i j, i ≠ j → i ≠ σ j → A i j = 0) (a : ι) :
    A.det = (∏ j, A j j) +
      ((Perm.sign σ : ℤ) : ℝ) * (∏ j, A (σ j) j) := by
  have hσ : σ ≠ 1 := by
    intro h
    exact cycle_no_fixed σ hc hn a (by simp [h])
  have hterm (π : Perm ι) :
      ((Perm.sign π : ℤ) : ℝ) * (∏ j, A (π j) j) =
        (if π = 1 then (∏ j, A j j) else 0) +
        (if π = σ then ((Perm.sign σ : ℤ) : ℝ) *
          (∏ j, A (σ j) j) else 0) := by
    by_cases hp : π = 1
    · subst π
      simp [hσ.symm]
    by_cases hp' : π = σ
    · subst π
      simp [hσ]
    simp only [hp, hp', ite_false, zero_add]
    suffices hz : (∏ j, A (π j) j) = 0 by simp [hz]
    by_contra hz
    have hs : ∀ j, π j = j ∨ π j = σ j := by
      intro j
      by_contra h
      push_neg at h
      exact hz (Finset.prod_eq_zero (Finset.mem_univ j) (hA (π j) j h.1 h.2))
    exact hp' (one_way_permutation σ π hc hn hs hp)
  calc
    A.det = ∑ π : Perm ι,
        ((if π = 1 then (∏ j, A j j) else 0) +
        (if π = σ then ((Perm.sign σ : ℤ) : ℝ) *
          (∏ j, A (σ j) j) else 0)) := by
      rw [Matrix.det_apply']
      exact Finset.sum_congr rfl (fun π _ => hterm π)
    _ = _ := by
      rw [Finset.sum_add_distrib]
      simp

def cyclicBidiagonal (σ : Perm ι) (a b : ι → ℝ) : Matrix ι ι ℝ :=
  fun i j => if i = j then a j else if i = σ j then -b j else 0

theorem cycle_sign (σ : Perm ι)
    (hc : σ.IsCycleOn (Finset.univ : Finset ι))
    (hn : 3 ≤ Fintype.card ι) (a : ι) :
    ((Perm.sign σ : ℤ) : ℝ) = -(-1 : ℝ) ^ Fintype.card ι := by
  have hcycle : σ.IsCycle := by
    refine ⟨a, cycle_no_fixed σ hc hn a, ?_⟩
    intro b _
    exact hc.2 (Finset.mem_univ a) (Finset.mem_univ b)
  have hs : σ.support = Finset.univ := by
    ext i
    simp [Perm.mem_support, cycle_no_fixed σ hc hn i]
  have hh := hcycle.sign
  rw [hs, Finset.card_univ] at hh
  have hh' := congrArg (fun x : ℤˣ => ((x : ℤ) : ℝ)) hh
  simpa using hh'

theorem cyclicBidiagonal_det (σ : Perm ι)
    (hc : σ.IsCycleOn (Finset.univ : Finset ι))
    (hn : 3 ≤ Fintype.card ι) (a b : ι → ℝ) (i : ι) :
    (cyclicBidiagonal σ a b).det = (∏ j, a j) - ∏ j, b j := by
  have hs : ∀ k j, k ≠ j → k ≠ σ j → cyclicBidiagonal σ a b k j = 0 := by
    intro k j hk hkj
    simp [cyclicBidiagonal, hk, hkj]
  rw [cyclic_bidiagonal_determinant (cyclicBidiagonal σ a b) σ hc hn hs i]
  simp only [cyclicBidiagonal, ite_true, cycle_no_fixed σ hc hn, ite_false]
  rw [cycle_sign σ hc hn i, Finset.prod_neg]
  have hh : (-1 : ℝ) ^ Fintype.card ι * (-1) ^ Fintype.card ι = 1 := by
    rw [← mul_pow]
    norm_num
  simp only [Finset.card_univ]
  rw [neg_mul, ← mul_assoc, hh, one_mul]
  rfl

theorem cyclic_gram_posDef (σ : Perm ι)
    (hc : σ.IsCycleOn (Finset.univ : Finset ι))
    (hn : 3 ≤ Fintype.card ι) (a b : ι → ℝ) (i : ι)
    (hab : (∏ j, a j) ≠ ∏ j, b j) :

def cycleTilt (A : Matrix ι ι ℝ) (σ : Perm ι) (q : ι → ℝ) : Matrix ι ι ℝ :=
  fun i j => (if i = σ j then q j else 1) *
    (if j = σ i then (q i)⁻¹ else 1) * A i j

theorem cycleTilt_prod (A : Matrix ι ι ℝ) (σ π : Perm ι) (q : ι → ℝ) :
    (∏ j, cycleTilt A σ q (π j) j) =
      (∏ j, if π j = σ j then q j else 1) *
      (∏ j, if π (σ j) = j then (q j)⁻¹ else 1) *
      (∏ j, A (π j) j) := by
  have hinv (j : ι) : π⁻¹ j = σ j ↔ π (σ j) = j := by
    constructor
    · intro h
      have hh := congrArg π h
      simpa using hh.symm
    · intro h
      apply π.injective
      simpa using h.symm
  have hb : (∏ j, if j = σ (π j) then (q (π j))⁻¹ else 1) =
      ∏ j, if π (σ j) = j then (q j)⁻¹ else 1 := by
    calc
      _ = ∏ j, if π⁻¹ j = σ j then (q j)⁻¹ else 1 := by
        simpa using Equiv.prod_comp π (fun j => if π⁻¹ j = σ j then (q j)⁻¹ else 1)
      _ = _ := by simp_rw [hinv]
  simp only [cycleTilt, Finset.prod_mul_distrib]
  rw [hb]

theorem cycle_forward_ne_backward (σ : Perm ι)
    (hc : σ.IsCycleOn (Finset.univ : Finset ι))
    (hn : 3 ≤ Fintype.card ι) (i : ι) : σ i ≠ σ⁻¹ i := by
  intro h
  apply cycle_no_two σ hc hn i
  simpa using congrArg σ h

theorem permutation_term_multitilt (A : Matrix ι ι ℝ) (σ π : Perm ι)
    (hc : σ.IsCycleOn (Finset.univ : Finset ι))
    (hn : 3 ≤ Fintype.card ι)
    (hA : ∀ i j, i ≠ j → i ≠ σ j → i ≠ σ⁻¹ j → A i j = 0)
    (q : ι → ℝ) (hq : ∀ i, q i ≠ 0) (a : ι) :
    (∏ j, cycleTilt A σ q (π j) j) =
      (∏ j, A (π j) j) +
      (if π = σ then ((∏ j, q j) - 1) * ∏ j, A (π j) j else 0) +
      (if π = σ⁻¹ then ((∏ j, q j)⁻¹ - 1) * ∏ j, A (π j) j else 0) := by
  rw [cycleTilt_prod]
  have hinv : σ ≠ σ⁻¹ := by
    intro h
    exact cycle_forward_ne_backward σ hc hn a (congrArg (fun f : Perm ι => f a) h)
  by_cases hp : π = σ
  · subst π
    simp only [ite_true, hinv, ite_false, add_zero,
      cycle_no_two σ hc hn, Finset.prod_const_one, mul_one]
    ring
  by_cases hp' : π = σ⁻¹
  · subst π
    have hrev (j : ι) : σ⁻¹ j ≠ σ j := (cycle_forward_ne_backward σ hc hn j).symm
    simp only [hp, ite_false, add_zero, ite_true, hrev,
      Perm.inv_apply_self, Finset.prod_const_one, one_mul, Finset.prod_inv_distrib]
    ring
  simp only [hp, hp', ite_false, add_zero]
  by_cases hz : (∏ j, A (π j) j) = 0
  · simp [hz]
  have hs : ∀ j, π j = j ∨ π j = σ j ∨ π j = σ⁻¹ j := by
    intro j
    by_contra h
    push_neg at h
    exact hz (Finset.prod_eq_zero (Finset.mem_univ j)
      (hA (π j) j h.1 h.2.1 h.2.2))
  have hpairs (j : ι) := counted_edge_paired σ π hc hn hs hp hp' j
  have hb : (∏ j, if π (σ j) = j then (q j)⁻¹ else 1) =
      (∏ j, if π j = σ j then q j else 1)⁻¹ := by
    rw [← Finset.prod_inv_distrib]
    apply Finset.prod_congr rfl
    intro j _
    by_cases he : π j = σ j
    · simp [he, (hpairs j).mp he]
    · have he' : π (σ j) ≠ j := fun h => he ((hpairs j).mpr h)
      simp [he, he']
  have hnz : (∏ j, if π j = σ j then q j else 1) ≠ 0 := by
    apply Finset.prod_ne_zero_iff.mpr
    intro j _
    split_ifs <;> simp_all
  rw [hb, mul_inv_cancel₀ hnz, one_mul]

  simp

/-!
Exact spectral factorization of a shifted real symmetric matrix.

This file uses mathlib's proved finite-dimensional spectral theorem.  It does
not assert that a particular stochastic generator has the required spectrum;
that is the separate rate-defined Gram-matrix obligation.

open scoped BigOperators

theorem det_shift_eq_prod (H : Matrix ι ι ℝ) (hH : H.IsHermitian) (s : ℝ) :
    (s • (1 : Matrix ι ι ℝ) + H).det =
      ∏ i, (s + hH.eigenvalues i) := by
  have hdiag :
      star (hH.eigenvectorUnitary : Matrix ι ι ℝ) * H *
        (hH.eigenvectorUnitary : Matrix ι ι ℝ) =
      Matrix.diagonal hH.eigenvalues := by
    simpa using hH.star_mul_self_mul_eq_diagonal
  have hconj :
      star (hH.eigenvectorUnitary : Matrix ι ι ℝ) *
          (s • (1 : Matrix ι ι ℝ) + H) *
          (hH.eigenvectorUnitary : Matrix ι ι ℝ) =
      s • (1 : Matrix ι ι ℝ) + Matrix.diagonal hH.eigenvalues := by
    rw [Matrix.mul_add, Matrix.add_mul, Matrix.mul_smul, Matrix.mul_one,
      Matrix.smul_mul, unitary.coe_star_mul_self, hdiag]
  have hdet := congrArg Matrix.det hconj
  rw [Matrix.det_mul_right_comm, unitary.coe_star_mul_self, Matrix.one_mul] at hdet
  rw [Matrix.smul_one_eq_diagonal, Matrix.diagonal_add, Matrix.det_diagonal] at hdet
  simpa only [Matrix.smul_one_eq_diagonal] using hdet

/-- Evaluation of the characteristic polynomial of `-L` is `det(s I + L)`. -/
theorem charpoly_neg_eval (L : Matrix ι ι ℝ) (s : ℝ) :
    (-L).charpoly.eval s = (s • (1 : Matrix ι ι ℝ) + L).det := by
  rw [Matrix.charpoly, Matrix.eval_det, Matrix.matPolyEquiv_charmatrix]
  simp [Matrix.scalar, Matrix.smul_one_eq_diagonal, Pi.algebraMap_def]

theorem charpoly_neg_eq_spectral_product
    (H : Matrix ι ι ℝ) (hH : H.IsHermitian) :
    (-H).charpoly = ∏ i, (X + C (hH.eigenvalues i)) := by
  apply Polynomial.funext
  intro s
  rw [charpoly_neg_eval, det_shift_eq_prod H hH]
  simp only [Polynomial.eval_prod, Polynomial.eval_add, Polynomial.eval_X,
    Polynomial.eval_C]

/-- A pointwise determinant shift gives the exact polynomial factorization. -/
theorem polynomial_factorization_of_det_shift
    (L H : Matrix ι ι ℝ) (hH : H.IsHermitian) (d : ℝ)
    (hshift : ∀ s : ℝ,
      (s • (1 : Matrix ι ι ℝ) + H).det =
      (s • (1 : Matrix ι ι ℝ) + L).det + d) :
    (-L).charpoly + C d = ∏ i, (X + C (hH.eigenvalues i)) := by
  apply Polynomial.funext
  intro s
  simp only [Polynomial.eval_add, Polynomial.eval_C]
  rw [charpoly_neg_eval, ← hshift, det_shift_eq_prod H hH]
  simp only [Polynomial.eval_prod, Polynomial.eval_add, Polynomial.eval_X,
    Polynomial.eval_C]

/-!
Finite-dimensional determinant jet transfer. A matrix-vector residual that
vanishes through order two forces the same vanishing for the determinant,
provided one component of the vector is one at zero. This is an ordinary
calculus theorem, intended to connect actual Poisson correctors to the first
two characteristic-polynomial cumulants without assuming a Perron branch.
-/
namespace UnicyclicJets
noncomputable section
open scoped BigOperators
set_option maxHeartbeats 2000000

def ZeroJet2 (f : ℝ → ℝ) : Prop :=
  f 0 = 0 ∧ deriv f 0 = 0 ∧ deriv (deriv f) 0 = 0

theorem second_deriv_mul (a b : ℝ → ℝ)
    (ha : ContDiff ℝ 2 a) (hb : ContDiff ℝ 2 b) :
    deriv (deriv (fun x => a x*b x)) 0 =
      deriv (deriv a) 0*b 0 + 2*deriv a 0*deriv b 0 +
        a 0*deriv (deriv b) 0 := by
  have ha' : ContDiff ℝ 1 (deriv a) := (contDiff_succ_iff_deriv.mp ha).2.2
  have hb' : ContDiff ℝ 1 (deriv b) := (contDiff_succ_iff_deriv.mp hb).2.2
  have hfirst : deriv (fun x => a x*b x) =
      (fun x => deriv a x*b x+a x*deriv b x) := by
    funext x
    exact (((ha.differentiable (by norm_num) x).hasDerivAt).mul
      ((hb.differentiable (by norm_num) x).hasDerivAt)).deriv
  rw [hfirst]
  have h := ((((ha'.differentiable (by norm_num) 0).hasDerivAt).mul
      ((hb.differentiable (by norm_num) 0).hasDerivAt)).add
    (((ha.differentiable (by norm_num) 0).hasDerivAt).mul
      ((hb'.differentiable (by norm_num) 0).hasDerivAt))).deriv
  convert h using 1 ; ring

theorem zeroJet2_mul (a b : ℝ → ℝ)
    (ha : ContDiff ℝ 2 a) (hb : ContDiff ℝ 2 b) (hz : ZeroJet2 b) :
    ZeroJet2 (fun x => a x*b x) := by
  rcases hz with ⟨h0,h1,h2⟩
  refine ⟨by simp [h0], ?_, ?_⟩
  · rw [deriv_mul (ha.differentiable (by norm_num) 0) (hb.differentiable (by norm_num) 0)]
    simp [h0,h1]
  · rw [second_deriv_mul a b ha hb]
    simp [h0,h1,h2]

theorem zeroJet2_cancel (a b : ℝ → ℝ)
    (ha : ContDiff ℝ 2 a) (hb : ContDiff ℝ 2 b) (hb0 : b 0 = 1)
    (hz : ZeroJet2 (fun x => a x*b x)) : ZeroJet2 a := by
  rcases hz with ⟨h0,h1,h2⟩
  have ha0 : a 0 = 0 := by simpa [hb0] using h0
  have ha1 : deriv a 0 = 0 := by
    rw [deriv_mul (ha.differentiable (by norm_num) 0) (hb.differentiable (by norm_num) 0)] at h1
    simpa [hb0,ha0] using h1
  have ha2 : deriv (deriv a) 0 = 0 := by
    rw [second_deriv_mul a b ha hb] at h2
    simpa [hb0,ha0,ha1] using h2
  exact ⟨ha0,ha1,ha2⟩

theorem zeroJet2_sum {ι : Type*} (s : Finset ι) (f : ι → ℝ → ℝ)
    (hf : ∀ i ∈ s, ContDiff ℝ 2 (f i))
    (hz : ∀ i ∈ s, ZeroJet2 (f i)) :
    ZeroJet2 (fun x => ∑ i ∈ s, f i x) := by
  have hfirst : deriv (fun x => ∑ i ∈ s, f i x) =
      (fun x => ∑ i ∈ s, deriv (f i) x) := by
    funext x
    exact (HasDerivAt.sum (fun i hi => ((hf i hi).differentiable (by norm_num) x).hasDerivAt)).deriv
  refine ⟨?_, ?_, ?_⟩
  · exact Finset.sum_eq_zero (fun i hi => (hz i hi).1)
  · rw [hfirst]
    exact Finset.sum_eq_zero (fun i hi => (hz i hi).2.1)
  · rw [hfirst]
    have hf' : ∀ i ∈ s, ContDiff ℝ 1 (deriv (f i)) :=
      fun i hi => (contDiff_succ_iff_deriv.mp (hf i hi)).2.2
    have hh := (HasDerivAt.sum (fun i hi =>
      ((hf' i hi).differentiable (by norm_num) 0).hasDerivAt)).deriv
    rw [hh]
    exact Finset.sum_eq_zero (fun i hi => (hz i hi).2.2)

variable {ι : Type*} [Fintype ι] [DecidableEq ι]

theorem contDiff_det (A : ℝ → Matrix ι ι ℝ)
    (hA : ∀ i j, ContDiff ℝ 2 (fun x => A x i j)) :
    ContDiff ℝ 2 (fun x => (A x).det) := by
  simp only [Matrix.det_apply']
  exact ContDiff.sum (fun p _ =>
    contDiff_const.mul (contDiff_prod (fun j _ => hA (p j) j)))

theorem contDiff_adjugate (A : ℝ → Matrix ι ι ℝ)
    (hA : ∀ i j, ContDiff ℝ 2 (fun x => A x i j)) (i j : ι) :
    ContDiff ℝ 2 (fun x => (A x).adjugate i j) := by

  apply contDiff_det
  intro k l
  by_cases hk : k = j
  · subst k
    simp only [Matrix.updateRow_self]
    exact contDiff_const
  · simp only [Matrix.updateRow_ne hk]

    (A : ℝ → Matrix ι ι ℝ) (v : ℝ → ι → ℝ) (i : ι)

    (hvi : v 0 i = 1)
    (hz : ∀ j, ZeroJet2 (fun x => (A x).mulVec (v x) j)) :
    ZeroJet2 (fun x => (A x).det) := by
  have hleft : ZeroJet2 (fun x =>
      ∑ j, (A x).adjugate i j*((A x).mulVec (v x) j)) := by
    apply zeroJet2_sum
    · intro j _
      exact (contDiff_adjugate A hA i j).mul (contDiff_mulVec A v hA hv j)
    · intro j _
      exact zeroJet2_mul _ _ (contDiff_adjugate A hA i j)
        (contDiff_mulVec A v hA hv j) (hz j)
  have hid : (fun x => ∑ j, (A x).adjugate i j*((A x).mulVec (v x) j)) =
      (fun x => (A x).det*v x i) := by
    funext x
    change ((A x).adjugate).mulVec ((A x).mulVec (v x)) i = _
    rw [Matrix.mulVec_mulVec,Matrix.adjugate_mul,Matrix.smul_mulVec_assoc,Matrix.one_mulVec]
    rfl
  rw [hid] at hleft
  exact zeroJet2_cancel _ _ (contDiff_det A hA) (hv i) hvi hleft

/-!
Constructive finite cyclic recurrences.  These are actual constructions from
positive rates and a single-cycle permutation, not definitions that assume the
stationary or corrector conclusions.  This file is under active checking.

def transferTerm (σ : Perm ι) (ρ f : ι → ℝ) (k : ℕ) (i : ι) : ℝ :=
  (∏ j ∈ Finset.range k, ρ ((σ^j) i)) * f ((σ^k) i)

def green (σ : Perm ι) (ρ f : ι → ℝ) (n : ℕ) (i : ι) : ℝ :=
  ∑ k ∈ Finset.range n, transferTerm σ ρ f k i

omit [Fintype ι] [DecidableEq ι] in
theorem transfer_step [Fintype ι] [DecidableEq ι] (σ : Perm ι) (ρ f : ι → ℝ) (k : ℕ) (i : ι) :
    ρ i * transferTerm σ ρ f k (σ i) = transferTerm σ ρ f (k+1) i := by
  have hshift (j : ℕ) : (σ^j) (σ i) = (σ^(j+1)) i := by
    rw [pow_succ, Perm.mul_apply]
  simp only [transferTerm, Finset.prod_range_succ', pow_zero, Perm.one_apply]
  simp_rw [hshift]
  ring

theorem green_recurrence_raw (σ : Perm ι) (ρ f : ι → ℝ) (n : ℕ) (i : ι) :
    green σ ρ f n i - ρ i * green σ ρ f n (σ i) =
      f i-transferTerm σ ρ f n i := by
  have hm : ρ i * green σ ρ f n (σ i) =
      ∑ k ∈ Finset.range n, transferTerm σ ρ f (k+1) i := by
    rw [green, Finset.mul_sum]
    apply Finset.sum_congr rfl
    intro k hk
    exact transfer_step σ ρ f k i
  have hs := Finset.sum_range_succ (fun k => transferTerm σ ρ f k i) n
  have hs' := Finset.sum_range_succ' (fun k => transferTerm σ ρ f k i) n
  have hzero : transferTerm σ ρ f 0 i = f i := by simp [transferTerm]
  dsimp only [green]
  dsimp only [green] at hm
  linarith only [hm, hs, hs', hzero]

omit [Fintype ι] [DecidableEq ι] in
theorem orbit_product [Fintype ι] [DecidableEq ι] (σ : Perm ι)
    (hc : σ.IsCycleOn (Finset.univ : Finset ι)) (ρ : ι → ℝ) (i : ι) :
    (∏ k ∈ Finset.range (Fintype.card ι), ρ ((σ^k) i)) = ∏ j, ρ j := by

  · intro k hk
    exact Finset.mem_univ _
  · intro k hk l hl hkl
    have he := (hc.pow_apply_eq_pow_apply (Finset.mem_univ i)).mp hkl
    have hk' := Finset.mem_range.mp hk
    have hl' := Finset.mem_range.mp hl
    simpa only [Finset.card_univ, Nat.ModEq, Nat.mod_eq_of_lt hk', Nat.mod_eq_of_lt hl'] using he
  · intro j hj
    obtain ⟨k, hk, he⟩ := hc.exists_pow_eq (Finset.mem_univ i) (Finset.mem_univ j)
    exact ⟨k, Finset.mem_range.mpr (by simpa using hk), he⟩
  · intro k hk
    rfl

    (hc : σ.IsCycleOn (Finset.univ : Finset ι)) (f : ι → ℝ) (i : ι) :
    (∑ k ∈ Finset.range (Fintype.card ι), f ((σ^k) i)) = ∑ j, f j := by

  · intro k hk
    exact Finset.mem_univ _
  · intro k hk l hl hkl
    have he := (hc.pow_apply_eq_pow_apply (Finset.mem_univ i)).mp hkl
    have hk' := Finset.mem_range.mp hk
    have hl' := Finset.mem_range.mp hl
    simpa only [Finset.card_univ, Nat.ModEq, Nat.mod_eq_of_lt hk', Nat.mod_eq_of_lt hl'] using he
  · intro j hj
    obtain ⟨k, hk, he⟩ := hc.exists_pow_eq (Finset.mem_univ i) (Finset.mem_univ j)
    exact ⟨k, Finset.mem_range.mpr (by simpa using hk), he⟩
  · intro k hk
    rfl

theorem green_recurrence (σ : Perm ι)
    (hc : σ.IsCycleOn (Finset.univ : Finset ι)) (ρ f : ι → ℝ) (i : ι) :
    green σ ρ f (Fintype.card ι) i -
      ρ i * green σ ρ f (Fintype.card ι) (σ i) =
      (1-∏ j, ρ j)*f i := by
  rw [green_recurrence_raw]
  have hp : (σ^(Fintype.card ι)) i = i := by
    simpa using hc.pow_card_apply (Finset.mem_univ i)
  rw [transferTerm, orbit_product σ hc ρ i, hp]
  ring

omit [Fintype ι] [DecidableEq ι] in
theorem green_positive [Fintype ι] [DecidableEq ι] (σ : Perm ι) (ρ f : ι → ℝ)
    (hρ : ∀ i, 0 < ρ i) (hf : ∀ i, 0 < f i) (i : ι) :
    0 < green σ ρ f (Fintype.card ι) i := by
  apply Finset.sum_pos
  · intro k hk
    exact mul_pos (Finset.prod_pos (fun j hj => hρ _)) (hf _)
  · exact ⟨0, Finset.mem_range.mpr (Fintype.card_pos_iff.mpr ⟨i⟩)⟩

theorem exists_positive_cyclic_solution (σ : Perm ι)
    (hc : σ.IsCycleOn (Finset.univ : Finset ι)) (ρ f : ι → ℝ)
    (hρ : ∀ i, 0 < ρ i) (hf : ∀ i, 0 < f i)
    (hprod : (∏ i, ρ i) < 1) :
    ∃ x : ι → ℝ, (∀ i, 0 < x i) ∧ ∀ i, x i-ρ i*x (σ i)=f i := by
  let Z : ℝ := 1-∏ i, ρ i
  have hZ : 0 < Z := sub_pos.mpr hprod
  refine ⟨fun i => green σ ρ f (Fintype.card ι) i/Z, ?_, ?_⟩
  · intro i
    exact div_pos (green_positive σ ρ f hρ hf i) hZ
  · intro i
    have hh := green_recurrence σ hc ρ f i
    change green σ ρ f (Fintype.card ι) i/Z -
      ρ i*(green σ ρ f (Fintype.card ι) (σ i)/Z) = f i
    calc
      _ = (green σ ρ f (Fintype.card ι) i-
            ρ i*green σ ρ f (Fintype.card ι) (σ i))/Z := by ring
      _ = f i := by
        rw [hh]
        change Z*f i/Z=f i
        field_simp [ne_of_gt hZ]

/-- A normalized positive stationary flow is constructed from the rates. -/
theorem exists_stationary_cycle [Nonempty ι] (σ : Perm ι)
    (hc : σ.IsCycleOn (Finset.univ : Finset ι)) (u v : ι → ℝ)
    (hu : ∀ i, 0 < u i) (hv : ∀ i, 0 < v i)
    (hUV : (∏ i, v i) < ∏ i, u i) :
    ∃ (π : ι → ℝ) (J : ℝ), (∀ i, 0 < π i) ∧
      (∑ i, π i)=1 ∧ 0<J ∧ ∀ i, π i*u i-π (σ i)*v i=J := by
  have hup : 0 < ∏ i, u i := Finset.prod_pos (fun i hi => hu i)
  have hp : (∏ i, v i/u i) < 1 := by
    rw [Finset.prod_div_distrib]
    exact (div_lt_one hup).mpr hUV
  obtain ⟨x, hx, he⟩ := exists_positive_cyclic_solution σ hc
    (fun i => v i/u i) (fun i => 1/u i)
    (fun i => div_pos (hv i) (hu i)) (fun i => one_div_pos.mpr (hu i)) hp
  let Z := ∑ i, x i
  have hZ : 0 < Z := Finset.sum_pos (fun i hi => hx i) Finset.univ_nonempty
  have he' : ∀ i, u i*x i-v i*x (σ i)=1 := by
    intro i
    have hh := congrArg (fun z : ℝ => u i*z) (he i)
    field_simp [ne_of_gt (hu i)] at hh
    nlinarith only [hh]
  refine ⟨fun i => x i/Z, 1/Z, ?_, ?_, one_div_pos.mpr hZ, ?_⟩
  · intro i
    exact div_pos (hx i) hZ
  · simp only [div_eq_mul_inv, ← Finset.sum_mul]
    exact mul_inv_cancel₀ (ne_of_gt hZ)
  · intro i
    calc
      _ = (u i*x i-v i*x (σ i))/Z := by ring
      _ = 1/Z := by rw [he' i]

theorem exists_positive_gradient (σ : Perm ι)
    (hc : σ.IsCycleOn (Finset.univ : Finset ι)) (u v : ι → ℝ)
    (hu : ∀ i, 0 < u i) (hv : ∀ i, 0 < v i)
    (hUV : (∏ i, v i) < ∏ i, u i) (J : ℝ) (hJ : 0<J) :
    ∃ g : ι → ℝ, (∀ i, 0<g i) ∧
      ∀ i, u i*g i-v (σ⁻¹ i)*g (σ⁻¹ i)=J := by
  have hup : 0 < ∏ i, u i := Finset.prod_pos (fun i hi => hu i)
  have hp : (∏ i, v (σ⁻¹ i)/u i) < 1 := by
    rw [Finset.prod_div_distrib, Equiv.prod_comp σ⁻¹ v]
    exact (div_lt_one hup).mpr hUV
  obtain ⟨g, hg, he⟩ := exists_positive_cyclic_solution σ⁻¹ hc.inv
    (fun i => v (σ⁻¹ i)/u i) (fun i => J/u i)
    (fun i => div_pos (hv _) (hu i)) (fun i => div_pos hJ (hu i)) hp
  refine ⟨g, hg, ?_⟩
  intro i
  have hh := congrArg (fun z : ℝ => u i*z) (he i)
  field_simp [ne_of_gt (hu i)] at hh
  simp only [one_div] at hh
  nlinarith only [hh]

omit [Fintype ι] [DecidableEq ι] in
theorem gradient_sum_one [Fintype ι] [DecidableEq ι] (σ : Perm ι) (u v π g : ι → ℝ) (J : ℝ)
    (hJ : J ≠ 0) (hnorm : (∑ i, π i)=1)
    (hflux : ∀ i, π i*u i-π (σ i)*v i=J)
    (hrec : ∀ i, u i*g i-v (σ⁻¹ i)*g (σ⁻¹ i)=J) :
    (∑ i, g i)=1 := by
  have hleft : (∑ i, π i*(u i*g i-v (σ⁻¹ i)*g (σ⁻¹ i)))=J := by
    simp_rw [hrec]
    rw [← Finset.sum_mul, hnorm, one_mul]
  have hreindex :
      (∑ i, π i*(v (σ⁻¹ i)*g (σ⁻¹ i))) =
        ∑ i, π (σ i)*(v i*g i) := by
    simpa only [Perm.inv_apply_self] using
      (Equiv.sum_comp σ (fun i => π i*(v (σ⁻¹ i)*g (σ⁻¹ i)))).symm
  have hright : (∑ i, π i*(u i*g i-v (σ⁻¹ i)*g (σ⁻¹ i))) =
      J*(∑ i, g i) := by
    simp only [mul_sub, Finset.sum_sub_distrib]
    rw [hreindex, ← Finset.sum_sub_distrib]
    calc
      _ = ∑ i, (π i*u i-π (σ i)*v i)*g i := by
        apply Finset.sum_congr rfl
        intro i hi
        ring
      _ = ∑ i, J*g i := by simp_rw [hflux]
      _ = _ := (Finset.mul_sum _ _ _).symm
  apply mul_left_cancel₀ hJ
  simpa using hright.symm.trans hleft

/-- Explicitly integrate every zero-sum edge function around a finite cycle. -/
theorem exists_cyclic_coboundary [Nonempty ι] (σ : Perm ι)
    (hc : σ.IsCycleOn (Finset.univ : Finset ι)) (r : ι → ℝ)
    (hr : (∑ i, r i)=0) :
    ∃ h : ι → ℝ, ∀ i, h (σ i)-h i=r i := by
  let N : ℕ := Fintype.card ι
  let H : ι → ℝ := fun i => ∑ k ∈ Finset.range N, green σ (fun _ => 1) r k i
  have hN : (N : ℝ) ≠ 0 := by
    exact_mod_cast (ne_of_gt (Fintype.card_pos : 0 < Fintype.card ι))
  have hstep (k : ℕ) (i : ι) :
      green σ (fun _ => 1) r k (σ i)-green σ (fun _ => 1) r k i =
        r ((σ^k) i)-r i := by
    have hh := green_recurrence_raw σ (fun _ => 1) r k i
    simp only [transferTerm, Finset.prod_const_one, one_mul] at hh
    linarith only [hh]
  have hdiff (i : ι) : H (σ i)-H i = -(N : ℝ)*r i := by
    dsimp only [H]
    rw [← Finset.sum_sub_distrib]
    simp_rw [hstep]
    rw [Finset.sum_sub_distrib, orbit_sum σ hc r i, hr]
    simp only [Finset.sum_const, Finset.card_range, nsmul_eq_mul, zero_sub, neg_mul]
  refine ⟨fun i => -H i/(N : ℝ), ?_⟩
  intro i
  calc
    _ = -(H (σ i)-H i)/(N : ℝ) := by ring
    _ = r i := by rw [hdiff i]; field_simp [hN]

/-- Positive rates on a genuine finite cycle supply a centered Poisson
corrector for every winding increment of total one. -/
theorem exists_stationary_corrector [Nonempty ι] (σ : Perm ι)
    (hc : σ.IsCycleOn (Finset.univ : Finset ι)) (u v d : ι → ℝ)
    (hu : ∀ i, 0 < u i) (hv : ∀ i, 0 < v i)
    (hUV : (∏ i, v i) < ∏ i, u i) (hd : (∑ i, d i)=1) :
    ∃ (π : ι → ℝ) (J : ℝ) (h : ι → ℝ),
      (∀ i, 0<π i) ∧ (∑ i, π i)=1 ∧ 0<J ∧
      (∀ i, π i*u i-π (σ i)*v i=J) ∧ (∑ i, π i*h i)=0 ∧
      ∀ i, u i*(d i+h (σ i)-h i)-
        v (σ⁻¹ i)*(d (σ⁻¹ i)+h i-h (σ⁻¹ i))=J := by
  obtain ⟨π,J,hπ,hnorm,hJ,hflux⟩ := exists_stationary_cycle σ hc u v hu hv hUV
  obtain ⟨g,hg,hrec⟩ := exists_positive_gradient σ hc u v hu hv hUV J hJ
  have hgsum := gradient_sum_one σ u v π g J (ne_of_gt hJ) hnorm hflux hrec
  have hrsum : (∑ i, (g i-d i))=0 := by
    rw [Finset.sum_sub_distrib, hgsum, hd]
    ring
  obtain ⟨h,hdiff⟩ := exists_cyclic_coboundary σ hc (fun i => g i-d i) hrsum
  let μ : ℝ := ∑ i, π i*h i
  refine ⟨π,J,(fun i => h i-μ),hπ,hnorm,hJ,hflux,?_,?_⟩
  · simp only [mul_sub, Finset.sum_sub_distrib, ← Finset.sum_mul]
    rw [hnorm]
    dsimp [μ]
    ring
  · intro i
    have hf : d i+(h (σ i)-μ)-(h i-μ)=g i := by
      have hh := hdiff i
      linarith only [hh]
    have hb : d (σ⁻¹ i)+(h i-μ)-(h (σ⁻¹ i)-μ)=g (σ⁻¹ i) := by
      have hh := hdiff (σ⁻¹ i)
      simp only [Perm.apply_inv_self] at hh
      linarith only [hh]
    rw [hf,hb]
    exact hrec i

end
end UnicyclicConstruction

/-!
The finite flow inequality used in the proposed Markov-chain argument.
The inputs are positive forward/backward stationary edge flows of common net
flux, and corrected edge increments summing to one.  The theorem proves the
claimed inequality for their quadratic form.  Identification of these objects
with arbitrary ring rates and tilted-generator cumulants is NOT assumed proved.
-/
namespace UnicyclicFiniteTUR

theorem log_ratio_lower {x : ℝ} (hx : 1 ≤ x) :
    2*(x-1)/(x+1) ≤ Real.log x := by
  let f : ℝ → ℝ := fun y => Real.log y - 2*(y-1)/(y+1)
  have hd : ∀ y, 1 ≤ y →
      HasDerivAt f ((y-1)^2/(y*(y+1)^2)) y := by
    intro y hy
    have hy0 : 0 < y := lt_of_lt_of_le (by norm_num) hy
    have hy1 : y+1 ≠ 0 := ne_of_gt (by linarith)
    have hh := (Real.hasDerivAt_log (ne_of_gt hy0)).sub
      ((((hasDerivAt_id y).sub_const 1).const_mul 2).div
        ((hasDerivAt_id y).add_const 1) hy1)
    convert hh using 1 ; dsimp [f] ; field_simp ; ring
  have hm : MonotoneOn f (Set.Ici (1 : ℝ)) := by
    apply monotoneOn_of_deriv_nonneg (convex_Ici (1 : ℝ))
    · intro y hy
      exact (hd y hy).continuousAt.continuousWithinAt
    · intro y hy
      exact (hd y (interior_subset hy)).differentiableAt.differentiableWithinAt
    · intro y hy
      have hy' : 1 ≤ y := interior_subset hy
      rw [(hd y hy').deriv]
      apply div_nonneg (sq_nonneg _)
      exact mul_nonneg (by linarith) (sq_nonneg _)
  have hh := hm (by simp : (1 : ℝ) ∈ Set.Ici (1 : ℝ)) hx hx
  simp only [f, Real.log_one, sub_self, mul_zero, zero_div, sub_zero] at hh
  linarith only [hh]

theorem flow_log_lower {a b J : ℝ} (ha : 0 < a) (hb : 0 < b)
    (hJ : 0 < J) (hflux : a-b=J) :
    2*J/(a+b) ≤ Real.log (a/b) := by
  have hratio : 1 ≤ a/b := (le_div_iff₀ hb).mpr (by linarith)
  have hh := log_ratio_lower hratio
  have hab : a+b ≠ 0 := ne_of_gt (add_pos ha hb)
  have hid : 2*(a/b-1)/(a/b+1) = 2*J/(a+b) := by
    rw [← hflux]
    field_simp
  rwa [hid] at hh

theorem finite_flow_tur_product {ι : Type*} [Fintype ι]
    (a b g : ι → ℝ) (J : ℝ)
    (ha : ∀ i, 0 < a i) (hb : ∀ i, 0 < b i)
    (hJ : 0 < J) (hflux : ∀ i, a i-b i=J)
    (hg : (∑ i, g i) = 1) :
    2*J ≤ (∑ i, (a i+b i)*g i^2) *
      (∑ i, Real.log (a i/b i)) := by
  have hw : ∀ i, 0 < a i+b i := fun i => add_pos (ha i) (hb i)
  have hc2 : 0 ≤ ∑ i, (a i+b i)*g i^2 := by
    apply Finset.sum_nonneg
    intro i hi
    exact mul_nonneg (le_of_lt (hw i)) (sq_nonneg _)
  have hcs : 1 ≤ (∑ i, (a i+b i)*g i^2) * ∑ i, (a i+b i)⁻¹ := by
    have hh := Finset.sum_sq_le_sum_mul_sum_of_sq_eq_mul
      (Finset.univ : Finset ι) (r := g)
      (f := fun i => (a i+b i)*g i^2) (g := fun i => (a i+b i)⁻¹)
      (fun i hi => mul_nonneg (le_of_lt (hw i)) (sq_nonneg _))
      (fun i hi => le_of_lt (inv_pos.mpr (hw i)))
      (fun i hi => by field_simp [ne_of_gt (hw i)])
    simpa [hg] using hh
  have hlog : 2*J*(∑ i, (a i+b i)⁻¹) ≤
      ∑ i, Real.log (a i/b i) := by
    rw [Finset.mul_sum]
    apply Finset.sum_le_sum
    intro i hi
    simpa only [div_eq_mul_inv] using
      flow_log_lower (ha i) (hb i) hJ (hflux i)
  have hmul := mul_le_mul_of_nonneg_left hlog hc2
  have htwoJ : 0 ≤ 2*J := by positivity
  have hmul2 := mul_le_mul_of_nonneg_left hcs htwoJ
  nlinarith only [hmul, hmul2]

theorem finite_flow_tur {ι : Type*} [Fintype ι]
    (a b g : ι → ℝ) (J : ℝ)
    (ha : ∀ i, 0 < a i) (hb : ∀ i, 0 < b i)
    (hJ : 0 < J) (hflux : ∀ i, a i-b i=J)
    (hg : (∑ i, g i) = 1) :
    2/(∑ i, Real.log (a i/b i)) ≤
      (∑ i, (a i+b i)*g i^2)/J := by
  have hp := finite_flow_tur_product a b g J ha hb hJ hflux hg
  have hc2 : 0 ≤ ∑ i, (a i+b i)*g i^2 := by
    exact Finset.sum_nonneg (fun i hi =>
      mul_nonneg (le_of_lt (add_pos (ha i) (hb i))) (sq_nonneg _))
  have hA : 0 < ∑ i, Real.log (a i/b i) := by
    by_contra hn
    have hn' := le_of_not_gt hn
    have hprod := mul_nonpos_of_nonneg_of_nonpos hc2 hn'
    linarith only [hp, hprod, hJ]
  exact (div_le_div_iff₀ hA hJ).mpr hp

end
end UnicyclicFiniteTUR

/-! Finite Poisson solvability and perturbation coefficients for actual cyclic
rates. These definitions use the generator acting on functions (the transpose
of the column generator used in the determinant development). -/

def generator (σ : Perm ι) (u v f : ι → ℝ) (i : ι) : ℝ :=
  u i*(f (σ i)-f i)+v (σ⁻¹ i)*(f (σ⁻¹ i)-f i)

def drift (σ : Perm ι) (u v d : ι → ℝ) (i : ι) : ℝ :=
  u i*d i-v (σ⁻¹ i)*d (σ⁻¹ i)

def tiltFirst (σ : Perm ι) (u v d f : ι → ℝ) (i : ι) : ℝ :=
  u i*d i*f (σ i)-v (σ⁻¹ i)*d (σ⁻¹ i)*f (σ⁻¹ i)

def tiltSecondOne (σ : Perm ι) (u v d : ι → ℝ) (i : ι) : ℝ :=
  u i*d i^2+v (σ⁻¹ i)*d (σ⁻¹ i)^2

omit [Fintype ι] [DecidableEq ι] in
theorem weighted_gradient_sum [Fintype ι] [DecidableEq ι] (σ : Perm ι) (u v π g : ι → ℝ) (J : ℝ)
    (hflux : ∀ i, π i*u i-π (σ i)*v i=J) :
    (∑ i, π i*(u i*g i-v (σ⁻¹ i)*g (σ⁻¹ i))) = J*(∑ i, g i) := by

      (∑ i, π i*(v (σ⁻¹ i)*g (σ⁻¹ i))) =
        ∑ i, π (σ i)*(v i*g i) := by
    simpa only [Perm.inv_apply_self] using
      (Equiv.sum_comp σ (fun i => π i*(v (σ⁻¹ i)*g (σ⁻¹ i)))).symm
  simp only [mul_sub, Finset.sum_sub_distrib]
  rw [hreindex, ← Finset.sum_sub_distrib]
  calc
    _ = ∑ i, (π i*u i-π (σ i)*v i)*g i := by
      apply Finset.sum_congr rfl
      intro i hi
      ring
    _ = ∑ i, J*g i := by simp_rw [hflux]
    _ = _ := (Finset.mul_sum _ _ _).symm

theorem stationary_generator_zero (σ : Perm ι) (u v π f : ι → ℝ) (J : ℝ)
    (hflux : ∀ i, π i*u i-π (σ i)*v i=J) :

  have he (i : ι) :

  simp_rw [he] at hh

theorem exists_signed_cyclic_solution (σ : Perm ι)
    (hc : σ.IsCycleOn (Finset.univ : Finset ι)) (ρ f : ι → ℝ)
    (hprod : (∏ i, ρ i)<1) :
    ∃ x : ι → ℝ, ∀ i, x i-ρ i*x (σ i)=f i := by

  have hZ : 0<Z := sub_pos.mpr hprod

    _ = (green σ ρ f (Fintype.card ι) i-

      field_simp [ne_of_gt hZ]

/-- Every zero-stationary-mean forcing has a centered Poisson solution,
constructed through a finite cyclic Green sum. -/
theorem exists_centered_poisson [Nonempty ι] (σ : Perm ι)
    (hc : σ.IsCycleOn (Finset.univ : Finset ι)) (u v π f : ι → ℝ) (J : ℝ)
    (hu : ∀ i, 0<u i) (_hv : ∀ i, 0<v i)
    (hUV : (∏ i, v i)<∏ i, u i)
    (hnorm : (∑ i, π i)=1) (hJ : J≠0)
    (hflux : ∀ i, π i*u i-π (σ i)*v i=J)
    (hf : (∑ i, π i*f i)=0) :
    ∃ h : ι → ℝ, (∑ i, π i*h i)=0 ∧ ∀ i, generator σ u v h i=f i := by
  have hup : 0<∏ i, u i := Finset.prod_pos (fun i hi => hu i)
  have hp : (∏ i, v (σ⁻¹ i)/u i)<1 := by
    rw [Finset.prod_div_distrib, Equiv.prod_comp σ⁻¹ v]
    exact (div_lt_one hup).mpr hUV
  obtain ⟨g,he⟩ := exists_signed_cyclic_solution σ⁻¹ hc.inv
    (fun i => v (σ⁻¹ i)/u i) (fun i => f i/u i) hp
  have hrec : ∀ i, u i*g i-v (σ⁻¹ i)*g (σ⁻¹ i)=f i := by
    intro i
    have hh := congrArg (fun z : ℝ => u i*z) (he i)
    field_simp [ne_of_gt (hu i)] at hh
    simp only [one_div] at hh
    nlinarith only [hh]
  have hgsum : (∑ i, g i)=0 := by
    have hh := weighted_gradient_sum σ u v π g J hflux
    simp_rw [hrec] at hh
    rw [hf] at hh
    exact (mul_eq_zero.mp hh.symm).resolve_left hJ
  obtain ⟨h,hdiff⟩ := exists_cyclic_coboundary σ hc g hgsum
  let μ : ℝ := ∑ i, π i*h i
  refine ⟨fun i => h i-μ, ?_, ?_⟩
  · simp only [mul_sub, Finset.sum_sub_distrib, ← Finset.sum_mul]
    rw [hnorm]
    dsimp [μ]
    ring
  · intro i
    have hfwd : (h (σ i)-μ)-(h i-μ)=g i := by
      have hh := hdiff i
      linarith only [hh]
    have hback : (h (σ⁻¹ i)-μ)-(h i-μ)= -g (σ⁻¹ i) := by
      have hh := hdiff (σ⁻¹ i)
      simp only [Perm.apply_inv_self] at hh
      linarith only [hh]
    simp only [generator, hfwd, hback]
    nlinarith only [hrec i]

def secondCoefficient (σ : Perm ι) (u v d π h : ι → ℝ) : ℝ :=
  ∑ i, π i*(tiltSecondOne σ u v d i+2*tiltFirst σ u v d h i)

def edgeQuadratic (σ : Perm ι) (u v d π h : ι → ℝ) : ℝ :=
  ∑ i, (π i*u i+π (σ i)*v i)*(d i+h (σ i)-h i)^2

theorem secondCoefficient_eq_edgeQuadratic (σ : Perm ι)
    (u v d π h : ι → ℝ) (J : ℝ)
    (hflux : ∀ i, π i*u i-π (σ i)*v i=J)
    (hcenter : (∑ i, π i*h i)=0)
    (hcorr : ∀ i, generator σ u v h i+drift σ u v d i=J) :
    secondCoefficient σ u v d π h = edgeQuadratic σ u v d π h := by
  let K : ι → ℝ := fun i =>
    u i*(d i+h (σ i)-h i)^2+
      v (σ⁻¹ i)*(d (σ⁻¹ i)+h i-h (σ⁻¹ i))^2
  have hpoint (i : ι) :
      K i+2*h i*J =
        (tiltSecondOne σ u v d i+2*tiltFirst σ u v d h i)+
          generator σ u v (fun i => h i^2) i := by
    rw [← hcorr i]
    dsimp [K, generator, drift, tiltSecondOne, tiltFirst]
    ring
  have hs :
      (∑ i, (π i*K i+2*J*(π i*h i))) =
        ∑ i, (π i*(tiltSecondOne σ u v d i+2*tiltFirst σ u v d h i)+
          π i*generator σ u v (fun i => h i^2) i) := by
    apply Finset.sum_congr rfl
    intro i hi
    calc
      _ = π i*(K i+2*h i*J) := by ring
      _ = π i*((tiltSecondOne σ u v d i+2*tiltFirst σ u v d h i)+
            generator σ u v (fun i => h i^2) i) := by rw [hpoint i]
      _ = _ := by ring
  simp only [Finset.sum_add_distrib, ← Finset.mul_sum] at hs
  rw [hcenter, stationary_generator_zero σ u v π (fun i => h i^2) J hflux] at hs
  have hK : (∑ i, π i*K i)=secondCoefficient σ u v d π h := by
    dsimp only [secondCoefficient]
    linarith only [hs]
  have hreindex :
      (∑ i, π i*v (σ⁻¹ i)*(d (σ⁻¹ i)+h i-h (σ⁻¹ i))^2) =
        ∑ i, π (σ i)*v i*(d i+h (σ i)-h i)^2 := by
    simpa only [Perm.inv_apply_self] using
      (Equiv.sum_comp σ (fun i =>
        π i*v (σ⁻¹ i)*(d (σ⁻¹ i)+h i-h (σ⁻¹ i))^2)).symm
  have hKedge : (∑ i, π i*K i)=edgeQuadratic σ u v d π h := by
    dsimp only [K, edgeQuadratic]
    simp only [mul_add, add_mul, Finset.sum_add_distrib, ← mul_assoc]
    rw [hreindex]
  exact hK.symm.trans hKedge

/-- Construct genuine first and second perturbation coefficients from the
rates, including the centered first and second vector coefficients. -/
theorem exists_two_coefficients [Nonempty ι] (σ : Perm ι)
    (hc : σ.IsCycleOn (Finset.univ : Finset ι)) (u v d : ι → ℝ)
    (hu : ∀ i, 0<u i) (hv : ∀ i, 0<v i)
    (hUV : (∏ i, v i)<∏ i, u i) (hd : (∑ i, d i)=1) :
    ∃ (π h k : ι → ℝ) (J C2 : ℝ),
      (∀ i, 0<π i) ∧ (∑ i, π i)=1 ∧ 0<J ∧
      (∀ i, π i*u i-π (σ i)*v i=J) ∧
      (∑ i, π i*h i)=0 ∧ (∑ i, π i*k i)=0 ∧
      (∀ i, generator σ u v h i+drift σ u v d i=J) ∧
      (∀ i, generator σ u v k i+2*tiltFirst σ u v d h i+
        tiltSecondOne σ u v d i=C2+2*J*h i) ∧
      C2=edgeQuadratic σ u v d π h := by
  obtain ⟨π,J,h,hπ,hnorm,hJ,hflux,hcenter,hcor⟩ :=
    exists_stationary_corrector σ hc u v d hu hv hUV hd
  have hfirst : ∀ i, generator σ u v h i+drift σ u v d i=J := by
    intro i
    have hh := hcor i
    dsimp [generator, drift]
    nlinarith only [hh]
  let C2 := secondCoefficient σ u v d π h
  let f : ι → ℝ := fun i => C2+2*J*h i-
    (tiltSecondOne σ u v d i+2*tiltFirst σ u v d h i)
  have hf : (∑ i, π i*f i)=0 := by
    have hs : (∑ i, π i*f i) =
        C2*(∑ i, π i)+2*J*(∑ i, π i*h i)-
          secondCoefficient σ u v d π h := by
      dsimp only [f, secondCoefficient]
      simp only [Finset.mul_sum, ← Finset.sum_add_distrib, ← Finset.sum_sub_distrib]
      apply Finset.sum_congr rfl
      intro i hi
      ring
    rw [hs, hnorm, hcenter]
    dsimp [C2]
    ring
  obtain ⟨k,hkcenter,hk⟩ := exists_centered_poisson σ hc u v π f J
    hu hv hUV hnorm (ne_of_gt hJ) hflux hf
  refine ⟨π,h,k,J,C2,hπ,hnorm,hJ,hflux,hcenter,hkcenter,hfirst,?_,?_⟩
  · intro i
    have hh := hk i
    dsimp only [f] at hh
    linarith only [hh]
  · exact secondCoefficient_eq_edgeQuadratic σ u v d π h J hflux hcenter hfirst

theorem stationary_flow_affinity [Fintype ι] [DecidableEq ι] (σ : Perm ι) (u v π : ι → ℝ)
    (hu : ∀ i, 0<u i) (hv : ∀ i, 0<v i) (hπ : ∀ i, 0<π i) :
    (∑ i, Real.log ((π i*u i)/(π (σ i)*v i))) =
      Real.log ((∏ i, u i)/(∏ i, v i)) := by
  have hterm (i : ι) :
      Real.log ((π i*u i)/(π (σ i)*v i)) =
        Real.log (u i/v i)+Real.log (π i)-Real.log (π (σ i)) := by
    rw [Real.log_div (mul_ne_zero (ne_of_gt (hπ i)) (ne_of_gt (hu i)))
      (mul_ne_zero (ne_of_gt (hπ _)) (ne_of_gt (hv i))),
      Real.log_mul (ne_of_gt (hπ i)) (ne_of_gt (hu i)),
      Real.log_mul (ne_of_gt (hπ _)) (ne_of_gt (hv i)),
      Real.log_div (ne_of_gt (hu i)) (ne_of_gt (hv i))]
    ring
  simp_rw [hterm]
  rw [Finset.sum_sub_distrib, Finset.sum_add_distrib, Equiv.sum_comp σ (fun i => Real.log (π i))]
  have hup : (∏ i, u i)≠0 := ne_of_gt (Finset.prod_pos (fun i hi => hu i))
  have hvp : (∏ i, v i)≠0 := ne_of_gt (Finset.prod_pos (fun i hi => hv i))
  rw [Real.log_div hup hvp,
    Real.log_prod Finset.univ u (fun i hi => ne_of_gt (hu i)),
    Real.log_prod Finset.univ v (fun i hi => ne_of_gt (hv i))]
  simp_rw [Real.log_div (ne_of_gt (hu _)) (ne_of_gt (hv _))]
  rw [Finset.sum_sub_distrib]
  ring

/-- The finite-flow TUR expressed using the actual rate-product affinity and
the corrected quadratic form. No uncertainty bound is a hypothesis. -/
theorem edgeQuadratic_tur (σ : Perm ι) (u v d π h : ι → ℝ) (J : ℝ)
    (hu : ∀ i, 0<u i) (hv : ∀ i, 0<v i) (hπ : ∀ i, 0<π i)
    (hJ : 0<J) (hflux : ∀ i, π i*u i-π (σ i)*v i=J)
    (hd : (∑ i, d i)=1) :
    2/Real.log ((∏ i, u i)/(∏ i, v i)) ≤ edgeQuadratic σ u v d π h/J := by
  have hg : (∑ i, (d i+h (σ i)-h i))=1 := by
    rw [Finset.sum_sub_distrib, Finset.sum_add_distrib, Equiv.sum_comp σ h, hd]
    ring
  have hh := UnicyclicFiniteTUR.finite_flow_tur
    (fun i => π i*u i) (fun i => π (σ i)*v i) (fun i => d i+h (σ i)-h i) J
    (fun i => mul_pos (hπ i) (hu i)) (fun i => mul_pos (hπ _) (hv i))
    hJ hflux hg
  rw [stationary_flow_affinity σ u v π hu hv hπ] at hh
  exact hh

end

theorem implicit_cumulant_equations_of_jets
    (P : Polynomial ℝ) (U V : ℝ) (f f1 f2 f3 : ℝ → ℝ)
    (hf1 : ∀ x, HasDerivAt f (f1 x) x)
    (hf2 : ∀ x, HasDerivAt f1 (f2 x) x)
    (hf3 : ∀ x, HasDerivAt f2 (f3 x) x)
    (hf0 : f 0 = 0)
    (hroot : ∀ x, P.eval (f x) = U*(Real.exp x-1)+V*(Real.exp (-x)-1)) :
    P.coeff 1 * f1 0 = U-V ∧
    P.coeff 1 * f2 0 + 2*P.coeff 2*(f1 0)^2 = U+V ∧
    P.coeff 1 * f3 0 + 6*P.coeff 2*f1 0*f2 0 +
      6*P.coeff 3*(f1 0)^3 = U-V := by
  have hrootfun : (fun x => P.eval (f x)) =
      (fun x => U*(Real.exp x-1)+V*(Real.exp (-x)-1)) := funext hroot
  have hfirst : ∀ x, P.derivative.eval (f x) * f1 x =
      U*Real.exp x - V*Real.exp (-x) := by
    intro x
    have hleft := (P.hasDerivAt (f x)).comp x (hf1 x)
    have hright := (((Real.hasDerivAt_exp x).sub_const 1).const_mul U).add
      (((hasDerivAt_exp_neg x).sub_const 1).const_mul V)
    simp only [Function.comp_def] at hleft
    rw [hrootfun] at hleft
    have heq := hleft.unique hright
    nlinarith only [heq]
  have hfirstfun : (fun x => P.derivative.eval (f x) * f1 x) =
      (fun x => U*Real.exp x - V*Real.exp (-x)) := funext hfirst
  have hsecond : ∀ x,
      P.derivative.eval (f x)*f2 x + P.derivative.derivative.eval (f x)*(f1 x)^2 =
      U*Real.exp x + V*Real.exp (-x) := by
    intro x
    have hleft := ((P.derivative.hasDerivAt (f x)).comp x (hf1 x)).mul (hf2 x)
    have hright := ((Real.hasDerivAt_exp x).const_mul U).sub
      ((hasDerivAt_exp_neg x).const_mul V)
    simp only [Function.comp_def] at hleft
    rw [hfirstfun] at hleft
    have heq := hleft.unique hright
    nlinarith only [heq]

      3*P.derivative.derivative.eval (f x)*f1 x*f2 x +
      P.derivative.derivative.derivative.eval (f x)*(f1 x)^3 =
      U*Real.exp x - V*Real.exp (-x) := by

    have hleft := (((P.derivative.hasDerivAt (f x)).comp x (hf1 x)).mul (hf3 x)).add
      (((P.derivative.derivative.hasDerivAt (f x)).comp x (hf1 x)).mul ((hf2 x).pow 2))

      ((hasDerivAt_exp_neg x).const_mul V)

    rw [hsecondfun] at hleft
    have heq := hleft.unique hright
    norm_num at heq
    nlinarith only [heq]

  have h2 := hsecond 0
  have h3 := hthird 0

  · exact h1
  · nlinarith only [h2]
  · nlinarith only [h3]

def characteristicResidual (P : Polynomial ℝ) (U V : ℝ) (f : ℝ → ℝ)
    (x : ℝ) : ℝ :=
  P.eval (f x) - (U*(Real.exp x-1)+V*(Real.exp (-x)-1))

theorem cumulant_equations_of_zero_residual_jet
    (P : Polynomial ℝ) (U V : ℝ) (f : ℝ → ℝ)
    (hf : ContDiff ℝ 2 f) (hf0 : f 0 = 0)
    (hres1 : deriv (characteristicResidual P U V f) 0 = 0)
    (hres2 : deriv (deriv (characteristicResidual P U V f)) 0 = 0) :
    P.coeff 1 * deriv f 0 = U-V ∧
    P.coeff 1 * deriv (deriv f) 0 + 2*P.coeff 2*(deriv f 0)^2 = U+V := by
  have hf' : ContDiff ℝ 1 (deriv f) := (contDiff_succ_iff_deriv.mp hf).2.2
  have hfirstfun : deriv (characteristicResidual P U V f) =
      (fun x => P.derivative.eval (f x)*deriv f x -
        (U*Real.exp x-V*Real.exp (-x))) := by
    funext x
    have hleft := (P.hasDerivAt (f x)).comp x
      (hf.differentiable (by norm_num) x).hasDerivAt
    have hright := (((Real.hasDerivAt_exp x).sub_const 1).const_mul U).add
      (((hasDerivAt_exp_neg x).sub_const 1).const_mul V)
    have hh := (hleft.sub hright).deriv
    simp only [Function.comp_def] at hh
    simpa only [characteristicResidual, mul_neg, sub_eq_add_neg,
      add_neg_cancel_right] using hh
  have h1 := hres1
  rw [hfirstfun] at h1
  have h2 := hres2
  rw [hfirstfun] at h2
  have hleft := ((P.derivative.hasDerivAt (f 0)).comp 0
      (hf.differentiable (by norm_num) 0).hasDerivAt).mul
        (hf'.differentiable (by norm_num) 0).hasDerivAt
  have hright := ((Real.hasDerivAt_exp 0).const_mul U).sub
      ((hasDerivAt_exp_neg 0).const_mul V)
  have hd := (hleft.sub hright).deriv
  simp only [Function.comp_def] at hd
  rw [hd] at h2
  norm_num [hf0, ← Polynomial.coeff_zero_eq_eval_zero,
    Polynomial.coeff_derivative] at h1 h2
  constructor <;> nlinarith only [h1,h2]

theorem implicit_cumulant_equations_of_local_jets
    (P : Polynomial ℝ) (U V : ℝ) (f f1 f2 f3 : ℝ → ℝ)
    (hf1 : ∀ᶠ x in 𝓝 (0 : ℝ), HasDerivAt f (f1 x) x)
    (hf2 : ∀ᶠ x in 𝓝 (0 : ℝ), HasDerivAt f1 (f2 x) x)
    (hf3 : HasDerivAt f2 (f3 0) 0)
    (hf0 : f 0 = 0)
    (hroot : ∀ᶠ x in 𝓝 (0 : ℝ),
      P.eval (f x) = U*(Real.exp x-1)+V*(Real.exp (-x)-1)) :
    P.coeff 1 * f1 0 = U-V ∧
    P.coeff 1 * f2 0 + 2*P.coeff 2*(f1 0)^2 = U+V ∧
    P.coeff 1 * f3 0 + 6*P.coeff 2*f1 0*f2 0 +
      6*P.coeff 3*(f1 0)^3 = U-V := by
  have hfirst : ∀ᶠ x in 𝓝 (0 : ℝ),
      P.derivative.eval (f x) * f1 x = U*Real.exp x - V*Real.exp (-x) := by
    filter_upwards [hroot.eventually_nhds, hf1] with x hxroot hxf1
    have hleft := (P.hasDerivAt (f x)).comp x hxf1
    have hright := (((Real.hasDerivAt_exp x).sub_const 1).const_mul U).add
      (((hasDerivAt_exp_neg x).sub_const 1).const_mul V)
    simp only [Function.comp_def] at hleft
    have heq := (hleft.congr_of_eventuallyEq
      (hxroot.mono (fun _ h => h.symm))).unique hright
    nlinarith only [heq]
  have hsecond : ∀ᶠ x in 𝓝 (0 : ℝ),
      P.derivative.eval (f x)*f2 x + P.derivative.derivative.eval (f x)*(f1 x)^2 =
      U*Real.exp x + V*Real.exp (-x) := by
    filter_upwards [hfirst.eventually_nhds, hf1, hf2] with x hxfirst hxf1 hxf2
    have hleft := ((P.derivative.hasDerivAt (f x)).comp x hxf1).mul hxf2
    have hright := ((Real.hasDerivAt_exp x).const_mul U).sub
      ((hasDerivAt_exp_neg x).const_mul V)
    simp only [Function.comp_def] at hleft
    have heq := (hleft.congr_of_eventuallyEq
      (hxfirst.mono (fun _ h => h.symm))).unique hright
    nlinarith only [heq]
  have hthird :
      P.derivative.eval (f 0)*f3 0 +
      3*P.derivative.derivative.eval (f 0)*f1 0*f2 0 +
      P.derivative.derivative.derivative.eval (f 0)*(f1 0)^3 =
      U*Real.exp 0 - V*Real.exp (-0) := by
    have hleft := (((P.derivative.hasDerivAt (f 0)).comp 0 hf1.self_of_nhds).mul hf3).add
      (((P.derivative.derivative.hasDerivAt (f 0)).comp 0 hf1.self_of_nhds).mul
        (hf2.self_of_nhds.pow 2))
    have hright := ((Real.hasDerivAt_exp 0).const_mul U).add
      ((hasDerivAt_exp_neg 0).const_mul V)
    simp only [Function.comp_def] at hleft
    have heq := (hleft.congr_of_eventuallyEq
      (hsecond.mono (fun _ h => h.symm))).unique hright
    norm_num at heq ⊢
    nlinarith only [heq]
  have h1 := hfirst.self_of_nhds
  have h2 := hsecond.self_of_nhds
  have h3 := hthird
  norm_num [hf0, ← Polynomial.coeff_zero_eq_eval_zero,
    Polynomial.coeff_derivative] at h1 h2 h3
  refine ⟨h1, ?_, ?_⟩
  · nlinarith only [h2]
  · nlinarith only [h3]

theorem implicit_cumulant_equations_local
    (P : Polynomial ℝ) (U V : ℝ) (f : ℝ → ℝ)
    (hf : ContDiffAt ℝ 3 f 0) (hf0 : f 0 = 0)
    (hroot : ∀ᶠ x in 𝓝 (0 : ℝ),
      P.eval (f x) = U*(Real.exp x-1)+V*(Real.exp (-x)-1)) :
    P.coeff 1 * deriv f 0 = U-V ∧
    P.coeff 1 * deriv (deriv f) 0 + 2*P.coeff 2*(deriv f 0)^2 = U+V ∧
    P.coeff 1 * deriv (deriv (deriv f)) 0 +
      6*P.coeff 2*deriv f 0*deriv (deriv f) 0 +
      6*P.coeff 3*(deriv f 0)^3 = U-V := by
  have hf' : ContDiffAt ℝ 2 (deriv f) 0 := by
    simpa only [deriv] using
      (hf.fderiv_right (m := 2) (by norm_num)).clm_apply contDiffAt_const
  have hf'' : ContDiffAt ℝ 1 (deriv (deriv f)) 0 := by
    simpa only [deriv] using
      (hf'.fderiv_right (m := 1) (by norm_num)).clm_apply contDiffAt_const
  apply implicit_cumulant_equations_of_local_jets P U V f
    (deriv f) (deriv (deriv f)) (deriv (deriv (deriv f)))
  · filter_upwards [hf.eventually (by norm_num)] with x hx
    exact (hx.differentiableAt (by norm_num)).hasDerivAt
  · filter_upwards [hf'.eventually (by norm_num)] with x hx
    exact (hx.differentiableAt (by norm_num)).hasDerivAt
  · exact (hf''.differentiableAt (by norm_num)).hasDerivAt
  · exact hf0
  · exact hroot

theorem polynomial_eval_contDiff (P : Polynomial ℝ) :
    ContDiff ℝ 3 (fun x : ℝ => P.eval x) := by
  induction P using Polynomial.induction_on' with
  | add p q hp hq => simpa using hp.add hq
  | monomial n a =>
    simpa using (contDiff_const (c := a)).mul (contDiff_id.pow n)

theorem exists_local_polynomial_branch
    (P : Polynomial ℝ) (g : ℝ → ℝ)
    (hP0 : P.eval 0 = 0) (hP1 : P.coeff 1 ≠ 0)
    (hg : ContDiffAt ℝ 3 g 0) (hg0 : g 0 = 0) :
    ∃ f : ℝ → ℝ, ContDiffAt ℝ 3 f 0 ∧ f 0 = 0 ∧
      ∀ᶠ x in 𝓝 (0 : ℝ), P.eval (f x) = g x := by
  have hPder : P.derivative.eval 0 ≠ 0 := by
    simpa [← Polynomial.coeff_zero_eq_eval_zero, Polynomial.coeff_derivative] using hP1
  have hPd : ContDiffAt ℝ 3 (fun x : ℝ => P.eval x) 0 :=
    (polynomial_eval_contDiff P).contDiffAt
  have hPf := (P.hasDerivAt 0).hasFDerivAt_equiv hPder
  have hn : (1 : WithTop ℕ∞) ≤ 3 := by norm_num
  let v : ℝ → ℝ := hPd.localInverse hPf hn
  have hv0 : v 0 = 0 := by
    simpa [v, hP0] using hPd.localInverse_apply_image hPf hn
  have hv : ContDiffAt ℝ 3 v 0 := by
    simpa [v, hP0] using hPd.to_localInverse hPf hn
  have hvr : ∀ᶠ y in 𝓝 (0 : ℝ), P.eval (v y) = y := by
    simpa [v, ContDiffAt.localInverse, hP0] using
      (hPd.hasStrictFDerivAt' hPf hn).eventually_right_inverse
  refine ⟨fun x => v (g x), ?_, ?_, ?_⟩
  · have hv' : ContDiffAt ℝ 3 v (g 0) := by simpa [hg0] using hv
    exact hv'.comp 0 hg
  · simpa [hg0] using hv0
  · have hgt : Tendsto g (𝓝 (0 : ℝ)) (𝓝 (0 : ℝ)) := by
      simpa [hg0] using hg.continuousAt.tendsto
    exact hgt.eventually hvr

theorem exists_local_winding_branch
    (P : Polynomial ℝ) (U V : ℝ)
    (hP0 : P.eval 0 = 0) (hP1 : P.coeff 1 ≠ 0) :
    ∃ f : ℝ → ℝ, ContDiffAt ℝ 3 f 0 ∧ f 0 = 0 ∧
      ∀ᶠ x in 𝓝 (0 : ℝ),

  apply exists_local_polynomial_branch P
    (fun x => U*(Real.exp x-1)+V*(Real.exp (-x)-1)) hP0 hP1
  · exact ((contDiff_const.mul (Real.contDiff_exp.sub contDiff_const)).add
      (contDiff_const.mul
        ((Real.contDiff_exp.comp contDiff_id.neg).sub contDiff_const))).contDiffAt
  · simp

theorem first_second_unique
    (a1 a2 U V C1 C2 : ℝ) (ha1 : a1 ≠ 0)
    (h1 : a1*C1 = U-V)
    (h2 : a1*C2+2*a2*C1^2 = U+V) :
    C1 = first a1 U V ∧ C2 = second a1 a2 U V := by
  have he1 := first_equation a1 U V ha1
  have he2 := second_equation a1 a2 U V ha1
  have hc1 : C1 = first a1 U V := mul_left_cancel₀ ha1 (h1.trans he1.symm)
  rw [hc1] at h2
  have hc2 : a1*C2 = a1*second a1 a2 U V := by linarith only [h2,he2]
  exact ⟨hc1,mul_left_cancel₀ ha1 hc2⟩

theorem rateMatrix_reversal [Fintype ι] [DecidableEq ι] (σ : Perm ι) (u v : ι → ℝ) (s : ℝ) :
    rateMatrix σ⁻¹ (fun j => v (σ⁻¹ j)) (fun j => u (σ⁻¹ j)) s =
      rateMatrix σ u v s := by
  ext i j
  have hf :
      (if j = σ⁻¹ i then -u (σ⁻¹ i) else 0) =
        (if i = σ j then -u j else 0) := by
    by_cases hij : i = σ j
    · subst i
      simp
    · have hji : j ≠ σ⁻¹ i := by
        intro h
        apply hij
        simpa using (congrArg σ h).symm
      simp [hij,hji]
  have hb :
      (if i = σ⁻¹ j then -v (σ⁻¹ j) else 0) =
        (if j = σ i then -v i else 0) := by
    by_cases hji : j = σ i
    · subst j
      simp
    · have hij : i ≠ σ⁻¹ j := by
        intro h
        apply hji
        simpa using (congrArg σ h).symm
      simp [hij,hji]
  simp only [rateMatrix,ringMatrix,inv_inv,Perm.inv_apply_self]
  rw [hf,hb]
  have hd : s+v (σ⁻¹ j)+u j = s+u j+v (σ⁻¹ j) := by ring
  rw [hd]
  ring

counted-edge tilted characteristic equation. -/
theorem winding_skewness_interval [Nonempty ι] (σ : Perm ι)
    (hc : σ.IsCycleOn (Finset.univ : Finset ι)) (hn : 3≤Fintype.card ι)
    (u v : ι → ℝ) (hu : ∀ i, 0<u i) (hv : ∀ i, 0<v i)
    (hUV : (∏ i, u i)≠∏ i, v i) (a : ι) (f : ℝ → ℝ)
    (hf : ContDiffAt ℝ 3 f 0) (hf0 : f 0=0)
    (hroot : ∀ᶠ s in 𝓝 (0 : ℝ), tiltedCharacteristic σ u v a s (f s)=0) :
    deriv f 0≠0 ∧
      -(1/16 : ℝ)≤deriv (deriv (deriv f)) 0/deriv f 0 ∧
      deriv (deriv (deriv f)) 0/deriv f 0≤1 := by
  have hp1 := rate_coefficient_pos σ hc hn u v hu hv hUV a
  have heq : ∀ᶠ s in 𝓝 (0 : ℝ),
      (ratePolynomial σ u v).eval (f s)=
        (∏ i, u i)*(Real.exp s-1)+(∏ i, v i)*(Real.exp (-s)-1) := by
    filter_upwards [hroot] with s hs
    rw [tiltedCharacteristic_formula σ hc hn] at hs
    exact sub_eq_zero.mp hs
  obtain ⟨h1,h2,h3⟩ := local_branch_derivatives (ratePolynomial σ u v)
    (∏ i, u i) (∏ i, v i) f (ne_of_gt hp1) hf hf0 heq
  have hnz : first ((ratePolynomial σ u v).coeff 1) (∏ i, u i) (∏ i, v i)≠0 :=
    div_ne_zero (sub_ne_zero.mpr hUV) (ne_of_gt hp1)
  rw [h1,h3]
  exact ⟨hnz,rate_skewness_bounds σ hc hn u v hu hv hUV a⟩

/-- Existence and the bound, obtained from the rates alone. -/
theorem exists_winding_branch_and_interval [Nonempty ι] (σ : Perm ι)
    (hc : σ.IsCycleOn (Finset.univ : Finset ι)) (hn : 3≤Fintype.card ι)
    (u v : ι → ℝ) (hu : ∀ i, 0<u i) (hv : ∀ i, 0<v i)
    (hUV : (∏ i, u i)≠∏ i, v i) (a : ι) :
    ∃ f : ℝ → ℝ, ContDiffAt ℝ 3 f 0 ∧ f 0=0 ∧
      (∀ᶠ s in 𝓝 (0 : ℝ), tiltedCharacteristic σ u v a s (f s)=0) ∧
      deriv f 0≠0 ∧
      -(1/16 : ℝ)≤deriv (deriv (deriv f)) 0/deriv f 0 ∧
      deriv (deriv (deriv f)) 0/deriv f 0≤1 := by
  have hp0 : (ratePolynomial σ u v).eval 0=0 := by
    rw [ratePolynomial,UnicyclicSpectral.charpoly_neg_eval]
    simp only [zero_smul,zero_add]
    exact rateMatrix_det_zero σ u v (fun i => (hu i).le) (fun i => (hv i).le) a
  have hp1 := rate_coefficient_pos σ hc hn u v hu hv hUV a
  obtain ⟨f,hf,hf0,hroot⟩ := UnicyclicImplicit.exists_local_winding_branch
    (ratePolynomial σ u v) (∏ i, u i) (∏ i, v i) hp0 (ne_of_gt hp1)
  have hz : ∀ᶠ s in 𝓝 (0 : ℝ), tiltedCharacteristic σ u v a s (f s)=0 := by
    filter_upwards [hroot] with s hs
    rw [tiltedCharacteristic_formula σ hc hn,hs,sub_self]
  exact ⟨f,hf,hf0,hz,winding_skewness_interval σ hc hn u v hu hv hUV a f hf hf0 hz⟩

#print axioms rate_skewness_bounds
#print axioms winding_skewness_interval
#print axioms exists_winding_branch_and_interval
end

section CheckedModule30

/-!
The two-state cycle, represented by two parallel reaction edges, has the
quadratic characteristic polynomial X² + (u0+u1+v0+v1)X.  This file checks
the stronger coefficient-ratio interval [1/4,1] for that small boundary
case. It is a completeness lemma, not the new arbitrary-size argument.
-/

namespace UnicyclicTwoState

noncomputable section

open UnicyclicCumulant

theorem quadratic_skewness_formula (a U V : ℝ) (ha : a ≠ 0) (hUV : U ≠ V) :
    third a 1 0 U V / first a U V =
      1-6*(U+V)/a^2+12*(U-V)^2/a^4 := by
  have hd : U-V ≠ 0 := sub_ne_zero.mpr hUV
  dsimp [third,second,first]
  field_simp [ha,hd]
  ring

theorem quadratic_scalar_interval (a α β : ℝ)
    (ha : 0<a) (hα : 0≤α) (hβ : 0≤β) (hab : 2*(α+β)≤a) :
    (1/4 : ℝ) ≤ 1-6*(α^2+β^2)/a^2+12*(α^2-β^2)^2/a^4 ∧
      1-6*(α^2+β^2)/a^2+12*(α^2-β^2)^2/a^4 ≤ 1 := by
  have hs : 0≤α+β := add_nonneg hα hβ
  have hp : 0≤α*β := mul_nonneg hα hβ
  have ha2 : 4*(α+β)^2≤a^2 := by
    have hh := mul_nonneg (sub_nonneg.mpr hab) (by positivity : 0≤a+2*(α+β))
    nlinarith only [hh]
  have hc : (α-β)^2≤(α+β)^2 := by nlinarith only [hp]
  have hlower := mul_nonneg (sub_nonneg.mpr ha2)
    (show 0≤a^2-4*(α-β)^2 by nlinarith only [ha2,hc])
  have hsum : 0≤α^2+β^2 := add_nonneg (sq_nonneg α) (sq_nonneg β)
  have hgap : 0≤a^2-2*(α^2+β^2) := by nlinarith only [ha2,hp,hsum]
  have hupper := mul_nonneg hsum hgap
  have hcross : 0≤α^2*β^2 := mul_nonneg (sq_nonneg α) (sq_nonneg β)
  have ha4 : 0<a^4 := pow_pos ha 4
  have heq : 1-6*(α^2+β^2)/a^2+12*(α^2-β^2)^2/a^4 =
      (a^4-6*(α^2+β^2)*a^2+12*(α^2-β^2)^2)/a^4 := by
    field_simp [ne_of_gt ha]
    ring
  rw [heq]
  constructor
  · apply (le_div_iff₀ ha4).mpr
    nlinarith only [hlower]
  · apply (div_le_iff₀ ha4).mpr
    nlinarith only [hupper,hcross]

theorem two_state_coefficient_skewness_interval
    (u0 u1 v0 v1 : ℝ) (hu0 : 0<u0) (hu1 : 0<u1)
    (hv0 : 0<v0) (hv1 : 0<v1) (hUV : u0*u1 ≠ v0*v1) :
    let a := u0+u1+v0+v1
    (1/4 : ℝ) ≤ third a 1 0 (u0*u1) (v0*v1) / first a (u0*u1) (v0*v1) ∧
      third a 1 0 (u0*u1) (v0*v1) / first a (u0*u1) (v0*v1) ≤ 1 := by
  dsimp only
  have ha : 0<u0+u1+v0+v1 := by positivity
  rw [quadratic_skewness_formula _ _ _ (ne_of_gt ha) hUV]
  have hα := Real.sqrt_nonneg (u0*u1)
  have hβ := Real.sqrt_nonneg (v0*v1)
  have hα2 := Real.sq_sqrt (mul_pos hu0 hu1).le
  have hβ2 := Real.sq_sqrt (mul_pos hv0 hv1).le
  have hαsum : 2*Real.sqrt (u0*u1)≤u0+u1 := by
    nlinarith [sq_nonneg (u0-u1)]
  have hβsum : 2*Real.sqrt (v0*v1)≤v0+v1 := by
    nlinarith [sq_nonneg (v0-v1)]
  have hab : 2*(Real.sqrt (u0*u1)+Real.sqrt (v0*v1))≤u0+u1+v0+v1 := by
    linarith only [hαsum,hβsum]
  simpa only [hα2,hβ2] using quadratic_scalar_interval
    (u0+u1+v0+v1) (Real.sqrt (u0*u1)) (Real.sqrt (v0*v1)) ha hα hβ hab

#print axioms two_state_coefficient_skewness_interval

section CheckedModule31

/-!
The actual counted-channel tilted determinant for the two-state ring.
Two parallel reaction channels must remain distinguished: the counting
factor is placed only on u0 and v0, not on an entire combined matrix entry.
This supplies the small-size boundary case of the winding theorem.
-/

open Filter Polynomial
open scoped Topology

namespace UnicyclicTwoState

noncomputable section

open UnicyclicCumulant UnicyclicImplicit

def twoStateMatrix (u0 u1 v0 v1 s z : ℝ) : Matrix (Fin 2) (Fin 2) ℝ :=
  !![z+u0+v1, -(u1+v0*Real.exp (-s));
     -(u0*Real.exp s+v1), z+u1+v0]

def quadraticRatePolynomial (u0 u1 v0 v1 : ℝ) : Polynomial ℝ :=
  X^2+C (u0+u1+v0+v1)*X

theorem twoState_characteristic_formula (u0 u1 v0 v1 s z : ℝ) :
    (twoStateMatrix u0 u1 v0 v1 s z).det =
      (quadraticRatePolynomial u0 u1 v0 v1).eval z -
        ((u0*u1)*(Real.exp s-1)+(v0*v1)*(Real.exp (-s)-1)) := by
  simp only [twoStateMatrix,Matrix.det_fin_two_of,
    quadraticRatePolynomial,Polynomial.eval_add,Polynomial.eval_pow,
    Polynomial.eval_X,Polynomial.eval_mul,Polynomial.eval_C]
  rw [Real.exp_neg]
  field_simp [Real.exp_ne_zero s]
  ring

theorem quadraticRatePolynomial_coefficients (u0 u1 v0 v1 : ℝ) :
    (quadraticRatePolynomial u0 u1 v0 v1).eval 0=0 ∧
    (quadraticRatePolynomial u0 u1 v0 v1).coeff 1=u0+u1+v0+v1 ∧
    (quadraticRatePolynomial u0 u1 v0 v1).coeff 2=1 ∧
    (quadraticRatePolynomial u0 u1 v0 v1).coeff 3=0 := by
  norm_num [quadraticRatePolynomial,Polynomial.coeff_add,
    Polynomial.coeff_C_mul,Polynomial.coeff_X_pow,Polynomial.coeff_X]

theorem two_state_winding_interval
    (u0 u1 v0 v1 : ℝ) (hu0 : 0<u0) (hu1 : 0<u1)
    (hv0 : 0<v0) (hv1 : 0<v1) (hUV : u0*u1 ≠ v0*v1)
    (f : ℝ → ℝ) (hf : ContDiffAt ℝ 3 f 0) (hf0 : f 0=0)
    (hroot : ∀ᶠ s in 𝓝 (0 : ℝ), (twoStateMatrix u0 u1 v0 v1 s (f s)).det=0) :
    deriv f 0 ≠ 0 ∧
      (1/4 : ℝ) ≤ deriv (deriv (deriv f)) 0 / deriv f 0 ∧
      deriv (deriv (deriv f)) 0 / deriv f 0 ≤ 1 := by
  have ha : 0<u0+u1+v0+v1 := by positivity
  obtain ⟨hp0,hp1,hp2,hp3⟩ := quadraticRatePolynomial_coefficients u0 u1 v0 v1
  have hp1ne : (quadraticRatePolynomial u0 u1 v0 v1).coeff 1≠0 := by
    rw [hp1]
    exact ne_of_gt ha
  have heq : ∀ᶠ s in 𝓝 (0 : ℝ),
      (quadraticRatePolynomial u0 u1 v0 v1).eval (f s) =
        (u0*u1)*(Real.exp s-1)+(v0*v1)*(Real.exp (-s)-1) := by
    filter_upwards [hroot] with s hs
    rw [twoState_characteristic_formula] at hs
    exact sub_eq_zero.mp hs
  obtain ⟨h1,h2,h3⟩ := local_branch_derivatives
    (quadraticRatePolynomial u0 u1 v0 v1) (u0*u1) (v0*v1)
    f hp1ne hf hf0 heq
  simp only [hp1,hp2,hp3] at h1 h2 h3
  rw [h1,h3]
  refine ⟨div_ne_zero (sub_ne_zero.mpr hUV) (ne_of_gt ha), ?_⟩
  exact two_state_coefficient_skewness_interval u0 u1 v0 v1 hu0 hu1 hv0 hv1 hUV

theorem exists_two_state_winding_branch_and_interval
    (u0 u1 v0 v1 : ℝ) (hu0 : 0<u0) (hu1 : 0<u1)
    (hv0 : 0<v0) (hv1 : 0<v1) (hUV : u0*u1 ≠ v0*v1) :
    ∃ f : ℝ → ℝ, ContDiffAt ℝ 3 f 0 ∧ f 0=0 ∧
      (∀ᶠ s in 𝓝 (0 : ℝ), (twoStateMatrix u0 u1 v0 v1 s (f s)).det=0) ∧
      deriv f 0 ≠ 0 ∧
      (1/4 : ℝ) ≤ deriv (deriv (deriv f)) 0 / deriv f 0 ∧
      deriv (deriv (deriv f)) 0 / deriv f 0 ≤ 1 := by
  have ha : 0<u0+u1+v0+v1 := by positivity
  obtain ⟨hp0,hp1,hp2,hp3⟩ := quadraticRatePolynomial_coefficients u0 u1 v0 v1
  have hp1ne : (quadraticRatePolynomial u0 u1 v0 v1).coeff 1≠0 := by
    rw [hp1]
    exact ne_of_gt ha
  obtain ⟨f,hf,hf0,hroot⟩ := exists_local_winding_branch
    (quadraticRatePolynomial u0 u1 v0 v1) (u0*u1) (v0*v1) hp0 hp1ne
  have hz : ∀ᶠ s in 𝓝 (0 : ℝ), (twoStateMatrix u0 u1 v0 v1 s (f s)).det=0 := by
    filter_upwards [hroot] with s hs
    rw [twoState_characteristic_formula,hs,sub_self]
  exact ⟨f,hf,hf0,hz,two_state_winding_interval u0 u1 v0 v1 hu0 hu1 hv0 hv1 hUV
    f hf hf0 hz⟩

/-!
For the one-state counted-event convention the winding process is the
difference of two Poisson counts. Its explicit cumulant generating
function has equal first and third derivatives. This is a boundary case,
not the novel arbitrary-cycle argument.

def poissonWinding (U V s : ℝ) : ℝ :=
  U*(Real.exp s-1)+V*(Real.exp (-s)-1)

theorem poisson_winding_derivatives (U V : ℝ) :
    deriv (poissonWinding U V) 0=U-V ∧
      deriv (deriv (deriv (poissonWinding U V))) 0=U-V := by
  have hf : ContDiff ℝ 3 (poissonWinding U V) := by
    exact (contDiff_const.mul (Real.contDiff_exp.sub contDiff_const)).add

        ((Real.contDiff_exp.comp contDiff_id.neg).sub contDiff_const))

    Polynomial.X U V (poissonWinding U V) hf (by simp [poissonWinding])
      (fun x => by simp [poissonWinding])
  norm_num [Polynomial.coeff_X] at h1 h2 h3
  exact ⟨h1,h3⟩

theorem poisson_winding_skewness (U V : ℝ) (hUV : U ≠ V) :
    deriv (poissonWinding U V) 0≠0 ∧
      deriv (deriv (deriv (poissonWinding U V))) 0 /
        deriv (poissonWinding U V) 0=1 := by
  obtain ⟨h1,h3⟩ := poisson_winding_derivatives U V
  rw [h1,h3]
