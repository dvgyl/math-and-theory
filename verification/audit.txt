PauliConditionalExact.every_pauli_channel (w : Fin 4 → ℝ) (hw : ∀ (i : Fin 4), 0 ≤ w i) (hsum : ∑ i, w i = 1) :
  ∃ a b,
    a ≠ b ∧
      (∀ (i : Fin 4), w a ≤ w i) ∧
        (∀ (i : Fin 4), i ≠ a → w b ≤ w i) ∧
          ∀ (γ : ℝ),
            1 ≤ γ →
              ConditionalHockey.conditionalCoefficient (PauliChannel.channel w hw hsum) γ =
                1 - 8 * w a * w b / (w a + w b)

def ConditionalHockey.conditionalRatioSet : {A B : Type} →
  [inst : Fintype A] →
    [inst_1 : DecidableEq A] →
      [inst_2 : Fintype B] → [inst_3 : DecidableEq B] → ConditionalHockey.QuantumChannel A B → ℝ → Set ℝ :=
fun {A B} [Fintype A] [DecidableEq A] [Fintype B] [DecidableEq B] N γ =>
  {r |
    ∃ R ρ σ,
      0 < ConditionalHockey.initialGap γ ρ σ ∧
        r = ConditionalHockey.finalGap N γ ρ σ / ConditionalHockey.initialGap γ ρ σ}
def ConditionalHockey.conditionalCoefficient : {A B : Type} →
  [inst : Fintype A] →
    [inst_1 : DecidableEq A] →
      [inst_2 : Fintype B] → [inst_3 : DecidableEq B] → ConditionalHockey.QuantumChannel A B → ℝ → ℝ :=
fun {A B} [Fintype A] [DecidableEq A] [Fintype B] [DecidableEq B] N γ =>
  sSup (ConditionalHockey.conditionalRatioSet N γ)
