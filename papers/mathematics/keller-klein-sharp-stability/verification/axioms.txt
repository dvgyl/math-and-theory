def KellerKlein.rademacherConcentration : {n : ℕ} → (Fin n → ℝ) → ℝ → ℝ :=
fun {n} c x => KellerKlein.cubeProbability fun ω => |KellerKlein.rademacherSum c ω| ≤ x
def KellerKlein.rademacherCDF : {n : ℕ} → (Fin n → ℝ) → ℝ → ℝ :=
fun {n} c x => KellerKlein.cubeProbability fun ω => KellerKlein.rademacherSum c ω ≤ x
def KellerKlein.cubeProbability : {n : ℕ} → (KellerKlein.BoolCube n → Prop) → ℝ :=
fun {n} p => ↑(Finset.filter p Finset.univ).card / 2 ^ n
def KellerKlein.rademacherSum : {n : ℕ} → (Fin n → ℝ) → KellerKlein.BoolCube n → ℝ :=
fun {n} c ω => ∑ i, c i * KellerKlein.rademacherSign (ω i)
def KellerKlein.H : ℝ → ℝ :=
fun a => 2 * KellerKlein.c a
def KellerKlein.c : ℝ → ℝ :=
fun a => ∫ (x : ℝ) in 0 ..a, KellerKlein.phi x
def KellerKlein.phi : ℝ → ℝ :=
fun x => KellerKlein.gaussian x * KellerKlein.phi0
def KellerKlein.phi0 : ℝ :=
1 / √(2 * Real.pi)
